/// COMMON before both: actual Manager/backend/executor/runtime/Workspace runner/SQL,
/// plus the SAME dispatcher wired to the native card's named row buttons.
@MainActor
enum TaskArtifactMetadataSelfTest {
    private enum FixtureError: Error { case unsafe, unexpectedCLI, noReview }
    private static func reached(_ predicate: @MainActor () -> Bool) async -> Bool {
        let until = ContinuousClock.now.advanced(by: .seconds(8))
        while ContinuousClock.now < until {
            if predicate() { return true }
            await Task.yield()
        }
        return predicate()
    }
    private static func canonical(_ receipt: ActionReceipt) throws -> ActionReceipt {
        let encoder = JSONEncoder(); encoder.dateEncodingStrategy = .iso8601
        let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
        return try decoder.decode(ActionReceipt.self, from: encoder.encode(receipt))
    }
    private struct Witness: Decodable { let tasks: [AgentTask] }
    private static func witnessTasks(_ url: URL) throws -> [AgentTask] {
        let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
        return try decoder.decode(Witness.self, from: Data(contentsOf: url)).tasks
    }
    private static func canonicalTasks(_ tasks: [AgentTask]) throws -> [AgentTask] {
        let encoder = JSONEncoder(); encoder.dateEncodingStrategy = .iso8601
        let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
        return try decoder.decode([AgentTask].self, from: encoder.encode(tasks))
    }
    @MainActor private final class CLI: WorkspaceCLIRunning {
        var calls: [[String]] = []
        var title = ""
        var body = ""
        let id: String
        let verified: Bool
        var beforeCreate: (() throws -> Void)?
        var beforeReadback: (() throws -> Void)?
        init(id: String, verified: Bool) { self.id = id; self.verified = verified }
        func run(_ arguments: [String], timeout: TimeInterval) async throws -> WorkspaceCLIOutput {
            calls.append(arguments)
            let value: [String: Any]
            if arguments.prefix(3) == ["docs", "documents", "create"], calls.count == 1 {
                try beforeCreate?()
                guard arguments.count == 5, arguments[3] == "--json",
                      let data = arguments[4].data(using: .utf8),
                      let object = try JSONSerialization.jsonObject(with: data) as? [String: Any],
                      let title = object["title"] as? String else { throw FixtureError.unexpectedCLI }
                self.title = title; value = ["documentId": id]
            } else if arguments.prefix(2) == ["docs", "+write"], calls.count == 2 {
                guard arguments.count == 6, arguments[2] == "--document", arguments[3] == id,
                      arguments[4] == "--text" else { throw FixtureError.unexpectedCLI }
                body = arguments[5]; value = [:]
            } else if arguments.prefix(3) == ["docs", "documents", "get"], calls.count == 3 {
                try beforeReadback?()
                guard arguments.count == 5, arguments[3] == "--params",
                      let data = arguments[4].data(using: .utf8),
                      let object = try JSONSerialization.jsonObject(with: data) as? [String: Any],
                      object["documentId"] as? String == id else { throw FixtureError.unexpectedCLI }
                value = ["documentId": id, "title": verified ? title : "Different readback title",
                    "body": ["content": [["paragraph": ["elements": [["textRun": ["content": body]]]]]]]]
            } else { throw FixtureError.unexpectedCLI }
            return WorkspaceCLIOutput(standardOutput: try JSONSerialization.data(withJSONObject: value),
                standardError: "", exitCode: 0)
        }
    }

    static func originalCases(root: URL, input: URL, text: String,
                              check: (Bool, String) -> Void) async throws {
        let safe = root.resolvingSymlinksInPath().standardizedFileURL
        let temp = FileManager.default.temporaryDirectory.resolvingSymlinksInPath().standardizedFileURL
        guard SelfTest.isRunning, safe.path.hasPrefix(temp.path + "/"),
              !safe.pathComponents.contains("Application Support"), PermissionGate.shared.pending == nil else {
            throw FixtureError.unsafe
        }
        let oldFake = AgentToolExecutor.fakeForTesting
        let oldFire = AgentToolExecutor.fireOverrideForTesting
        let oldPolicy = AgentToolExecutor.policyOverrideForTesting
        let oldCLI = AgentToolExecutor.workspaceCLIForTesting
        defer {
            AgentToolExecutor.fakeForTesting = oldFake
            AgentToolExecutor.fireOverrideForTesting = oldFire
            AgentToolExecutor.policyOverrideForTesting = oldPolicy
            AgentToolExecutor.workspaceCLIForTesting = oldCLI
            PermissionGate.shared.resetForRestartTesting()
        }
        AgentToolExecutor.fakeForTesting = nil
        AgentToolExecutor.fireOverrideForTesting = nil
        AgentToolExecutor.policyOverrideForTesting = .selfTest
        for mode in ["accepted", "readback-unverified", "start-rejected", "terminal-receipt-rejected",
                     "completion-journal-rejected", "terminal-task-rejected"] {
            let directory = safe.appendingPathComponent("artifact-" + mode, isDirectory: true)
            try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: false)
            let sql = TaskStore(root: directory); defer { sql.close() }
            let export = directory.appendingPathComponent("agent-tasks.json")
            let store = AgentTaskStore(fileURL: export, mirror: sql)
            guard store.allowsHarnessPersistence else { throw FixtureError.unsafe }
            let receiptFile = directory.appendingPathComponent("action-receipts.json")
            let retainedFile = directory.appendingPathComponent("retained-fired.json")
            let receipts = ActionReceiptStore(fileURL: receiptFile)
            let manager = AgentTaskManager(store: store, receiptStore: receipts)
            let cli = CLI(id: "owned-artifact-" + mode, verified: mode != "readback-unverified")
            AgentToolExecutor.workspaceCLIForTesting = cli
            let draft = "Draft fixture document"
            let approved = "Approved fixture document"
            let body = "Exact harmless fixture body. No account or external effect."
            var context: TaskJournalContext?
            var joined = false
            var retainedBytes: Data?
            var firedID: UUID?
            var delivered: [String] = []
            manager.announcementForTesting = { delivered.append($0) }
            manager.backendEntryForTesting = {
                context = TaskEventJournal.current
                if mode == "start-rejected" {
                    try sql.withConnection { try TaskStore.exec($0,
                        "CREATE TRIGGER IF NOT EXISTS refuse_artifact_start BEFORE INSERT ON task_event WHEN NEW.kind='toolStarted' BEGIN SELECT RAISE(ABORT,'artifact fixture'); END") }
                }
                if mode == "completion-journal-rejected" {
                    try sql.withConnection { try TaskStore.exec($0,
                        "CREATE TRIGGER IF NOT EXISTS refuse_artifact_finish BEFORE INSERT ON task_event WHEN NEW.kind='toolCompleted' BEGIN SELECT RAISE(ABORT,'artifact fixture'); END") }
                }
                if mode == "terminal-task-rejected" {
                    try sql.withConnection { try TaskStore.exec($0,
                        "CREATE TRIGGER IF NOT EXISTS refuse_artifact_task BEFORE UPDATE ON task WHEN NEW.status='completed' BEGIN SELECT RAISE(ABORT,'artifact fixture'); END") }
                }
            }
            cli.beforeCreate = {
                guard let context else { throw FixtureError.unsafe }
                let fired = ActionReceiptStore(fileURL: receiptFile).receipts.first {
                    $0.taskID == context.taskID && $0.taskAttempt == context.attempt && $0.status == .fired
                }
                let fresh = TaskStore(root: directory); defer { fresh.close() }
                let row = try fresh.load().first { $0.id == context.taskID }
                firedID = fired?.actionID
                joined = fired != nil && row?.status == .running
                    && row?.durability?.receiptIDs.contains(fired!.actionID.uuidString) == true
                    && row?.durability?.attempt == context.attempt
                    && fired?.taskLeaseOwner == String(ProcessInfo.processInfo.processIdentifier)
                    && fired?.preparedAction?.executionPlan.arguments == ["title": approved, "markdown": body]
            }
            cli.beforeReadback = {
                if mode == "terminal-receipt-rejected" {
                    retainedBytes = try Data(contentsOf: receiptFile)
                    try FileManager.default.moveItem(at: receiptFile, to: retainedFile)
                    try FileManager.default.createDirectory(at: receiptFile, withIntermediateDirectories: false)
                }
            }
            let task = manager.submit(objective: "Create the reviewed isolated fixture document",
                tool: "create_doc", arguments: ["title": draft, "markdown": body], source: "selftest")
            defer { _ = AgentArtifactLedger.take(taskID: task.id) }
            let parked = await reached { manager.task(id: task.id)?.status == .waitingForPermission }
            check(parked && cli.calls.isEmpty, "artifact control: unapproved task ran CLI (\(mode))")
            manager.restorePendingInteractions()
            guard let pending = PermissionGate.shared.pending, pending.taskID == task.id else { throw FixtureError.noReview }
            ToolCallReviewStore.shared.update(id: pending.id, field: "title", to: approved)
            for field in PermissionGate.shared.pendingReview?.fields ?? [] {
                ToolCallReviewStore.shared.confirm(id: pending.id, field: field.name)
            }
            check(PermissionGate.shared.pendingReview?.isReadyToRun == true
                && manager.task(id: task.id)?.pendingInteraction != nil,
                "artifact control: real saved edited review was not ready (\(mode))")
            manager.respondPermission(taskID: task.id, approved: true)
            let done = await reached {
                let status = manager.task(id: task.id)?.status
                return status != .queued && status != .running && status != .waitingForPermission
                    || (mode == "terminal-task-rejected" && manager.backendStartsForTesting == 2
                        && manager.workerBinding(taskID: task.id) == nil)
            }
            check(done, "artifact control: real task did not settle/release (\(mode))")
            let final = manager.task(id: task.id)
            let rows = try sql.load()
            let primary = rows.first { $0.id == task.id }
            let raw = [cli.id, "https://docs.google.com/document/d/\(cli.id)/edit"]
            let ledger = AgentArtifactLedger.peekRecords(taskID: task.id)
            SelfTest.diagnostic("TASK_ARTIFACT_PRODUCER_TRACE: mode=\(mode) calls=\(cli.calls.count) joined=\(joined) state=\(final?.status.rawValue ?? "missing") values=\(final?.artifacts.count ?? -1) kinds=\(final?.artifactRecords.map(\.kind).joined(separator: ",") ?? "missing") retainedRaw=\(ledger.count)")
            if mode == "start-rejected" {
                check(cli.calls.isEmpty && final?.status == .failed && final?.artifacts.isEmpty == true,
                    "artifact control: rejected start created/published a result")
                continue
            }
            check(joined && cli.calls.count == 3 && cli.title == approved && cli.body == body,
                "artifact control: backing lost approval/binding/one-create/readback (\(mode))")
            check(final?.durability?.receiptIDs.contains(firedID?.uuidString ?? "") == true,
                "artifact control: refused or admitted flow erased receipt join (\(mode))")
            if mode == "terminal-receipt-rejected" || mode == "completion-journal-rejected" || mode == "terminal-task-rejected" {
                check(final?.status != .completed && primary?.status != .completed
                    && !delivered.contains(where: { $0.contains("Created the Doc") }),
                    "artifact control: refused terminal admission delivered success (\(mode))")
                if mode != "terminal-task-rejected" {
                    check(ledger.map(\.value) == raw && final?.artifacts.isEmpty == true,
                        "artifact control: receipt/journal refusal erased raw related references or invented task success (\(mode))")
                }
                if mode == "terminal-receipt-rejected" {
                    check(try Data(contentsOf: retainedFile) == retainedBytes
                        && receipts.receipt(for: firedID!)?.status == .fired,
                        "artifact control: refused receipt write erased exact fired evidence")
                }
                if let final {
                    check(AgentActivityProjector.artifactHeading(for: final) == "Related results",
                        "artifact feature: failed/unadmitted task implied created success (\(mode))")
                }
                continue
            }
            let freshSQL = TaskStore(root: directory); defer { freshSQL.close() }
            let freshStore = AgentTaskStore(fileURL: export, mirror: freshSQL)
            let freshManager = AgentTaskManager(store: freshStore, receiptStore: receipts)
            let fresh = freshManager.task(id: task.id)
            check(final?.status == .completed && fresh?.status == .completed && freshManager.backendStartsForTesting == 0
                && fresh?.artifacts == raw && final?.artifacts == raw && ledger.isEmpty,
                "artifact control: accepted ordered paths/one consume/fresh no-replay changed (\(mode))")
            // SQL stores fractional dates; the actual export has the existing ISO8601 codec.
            // Canonicalize only that time domain; all ordered values/kinds/titles remain exact.
            check(try rows == freshSQL.load() && freshManager.historyReadFailure == nil
                && witnessTasks(export) == canonicalTasks(rows),
                "artifact control: decoded actual SQL/witness canonical equality changed (\(mode))")
            let saved = ActionReceiptStore(fileURL: receiptFile).receipts.first { $0.actionID == firedID }
            if mode == "accepted" {
                check(saved?.status == .completed && fresh?.artifactRecords.map(\.kind) == ["documentIDCreated", "documentCreated"]
                    && fresh?.artifactRecords.allSatisfy { $0.title == approved } == true,
                    "artifact feature: real approved document kind/title did not reach fresh canonical task")
                let physicalKinds = try freshSQL.withConnection { db in
                    try TaskStore.stringColumn(db, "SELECT kind FROM task_artifact ORDER BY ordinal")
                }
                check(physicalKinds == ["artifact-v1:documentIDCreated", "artifact-v1:documentCreated"],
                    "artifact feature: actual producer metadata did not occupy existing ordered SQL columns")
                if let fresh {
                    let cardRows = AgentWorkingCard.artifactRows(for: fresh)
                    check(cardRows.map(\.label) == [approved, approved] && cardRows.first?.openURL == nil
                        && cardRows.last?.openURL?.absoluteString == raw.last
                        && AgentActivityProjector.artifactHeading(for: fresh) == "What I made",
                        "artifact feature: actual card projection lost approved labels or honest Open/Copy semantics")
                }
            } else {
                check(saved?.status == .couldNotVerify && fresh?.artifactRecords.allSatisfy(\.isLegacy) == true,
                    "artifact control: unverified real readback promoted created metadata")
                if let fresh {
                    check(AgentActivityProjector.artifactHeading(for: fresh) == "Results"
                        && AgentWorkingCard.artifactRows(for: fresh).allSatisfy { $0.openURL == nil },
                        "artifact feature: couldNotVerify task implied created success or guessed an Open target")
                }
            }
            if let saved {
                let oldEncoder = JSONEncoder(); oldEncoder.dateEncodingStrategy = .iso8601
                let bytes = try oldEncoder.encode(saved)
                let object = try JSONSerialization.jsonObject(with: bytes) as? [String: Any]
                let oldDecoder = JSONDecoder(); oldDecoder.dateDecodingStrategy = .iso8601
                check(try oldDecoder.decode(ActionReceipt.self, from: bytes) == canonical(saved)
                    && object?["artifactRecords"] == nil && saved.artifacts == raw,
                    "artifact control: existing receipt codec/raw artifact/effect evidence changed")
            }
        }
        try await fileAndActionCases(root: safe, input: input, text: text, check: check)
        // Existing actual ACP peer cases are rerun unchanged under a disjoint safe subroot.
        let acpRoot = safe.appendingPathComponent("artifact-acp-control", isDirectory: true)
        try FileManager.default.createDirectory(at: acpRoot, withIntermediateDirectories: false)
        try await TaskACPUnverifiedReceiptSelfTest.originalCases(root: acpRoot, check: check)
        let acpSQL = TaskStore(root: acpRoot.appendingPathComponent("acp-unverified-accepted")); defer { acpSQL.close() }
        if let acp = try acpSQL.load().first {
            check(acp.artifactRecords.allSatisfy(\.isLegacy) && acp.artifacts == ["fixture-unverified-receipt"]
                && AgentActivityProjector.artifactHeading(for: acp) == "Results",
                "artifact feature: actual ACP couldNotVerify reply became created metadata")
        } else { check(false, "artifact control: actual accepted ACP row missing") }
    }

    private static func fileAndActionCases(root: URL, input: URL, text: String,
                                          check: (Bool, String) -> Void) async throws {
        let directory = root.appendingPathComponent("artifact-file-and-actions", isDirectory: true)
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: false)
        let sql = TaskStore(root: directory); defer { sql.close() }
        let store = AgentTaskStore(fileURL: directory.appendingPathComponent("agent-tasks.json"), mirror: sql)
        let receipts = ActionReceiptStore(fileURL: directory.appendingPathComponent("action-receipts.json"))
        let manager = AgentTaskManager(store: store, receiptStore: receipts)
        AgentToolExecutor.workspaceCLIForTesting = nil
        let read = manager.submit(objective: "Read the actual isolated artifact file", tool: "filesystem.read",
            arguments: ["path": input.path], source: "selftest")
        check(await reached { manager.task(id: read.id)?.status == .completed }, "artifact control: actual file read did not complete")
        guard let task = manager.task(id: read.id) else { throw FixtureError.unsafe }
        check(task.result?.contains(text) == true && task.artifacts == [input.path],
            "artifact control: file producer lost original content/path")
        check(task.artifactRecords.first?.kind == "filePathRead"
            && AgentActivityProjector.artifactHeading(for: task) == "What I read",
            "artifact feature: real file read was labelled created/unknown")
        var failure = task; failure.status = .failed; failure.failure = "Fixture later step failed."
        let row = AgentWorkingCard.artifactRows(for: failure).first!
        var opened: [URL] = []; var copied: [String] = []
        check(AgentWorkingCard.performArtifactAction(.open, task: failure, row: row,
            open: { opened.append($0); return true }, copy: { copied.append($0); return true })
            && opened == [input] && AgentActivityProjector.artifactHeading(for: failure) == "Related results",
            "artifact feature: actual path-only failure Open no-op or implied success")
        check(AgentWorkingCard.performArtifactAction(.copy, task: failure, row: row,
            open: { _ in false }, copy: { copied.append($0); return true }) && copied == [input.path],
            "artifact control: named Copy changed exact path")
        var other = failure; other.id = UUID().uuidString
        let before = opened.count
        check(!AgentWorkingCard.performArtifactAction(.open, task: other, row: row,
            open: { opened.append($0); return true }, copy: { _ in true }) && opened.count == before,
            "artifact control: task A row acted on task B")

        // Same actual consumer with an exact filename ending in a space. No path rewriting.
        let spacedFile = directory.appendingPathComponent("exact-name ")
        let spacedText = "Exact trailing-space artifact backing"
        try spacedText.write(to: spacedFile, atomically: true, encoding: .utf8)
        let spacedRead = manager.submit(objective: "Read the exact isolated trailing-space filename", tool: "filesystem.read",
            arguments: ["path": spacedFile.path], source: "selftest")
        check(await reached { manager.task(id: spacedRead.id)?.status == .completed },
            "artifact control: actual trailing-space file read did not complete")
        guard let spacedTask = manager.task(id: spacedRead.id),
              let spacedSaved = try sql.load().first(where: { $0.id == spacedRead.id }) else { throw FixtureError.unsafe }
        check(spacedTask.result?.contains(spacedText) == true && FileManager.default.fileExists(atPath: spacedFile.path),
            "artifact control: original trailing-space backing/input changed")
        check(spacedTask.artifacts == [spacedFile.path] && spacedSaved.artifactRecords == spacedTask.artifactRecords
            && spacedSaved.artifactRecords.first?.kind == "filePathRead"
            && spacedSaved.artifactRecords.first?.title == "exact-name ",
            "artifact feature: typed capture rewrote exact file location or emitted a duplicate trimmed path")
        let spacedRow = AgentWorkingCard.artifactRows(for: spacedSaved).first!
        var exactOpened: [URL] = []
        check(AgentWorkingCard.performArtifactAction(.open, task: spacedSaved, row: spacedRow,
            open: { exactOpened.append($0); return true }, copy: { _ in true }) && exactOpened == [spacedFile],
            "artifact feature: exact trailing-space producer location did not reach actual card Open backing")

        // Supplementary explicit producer-shaped action matrix; not a substitute for above real flows.
        let file = directory.appendingPathComponent("matrix.txt")
        try "Matrix backing".write(to: file, atomically: true, encoding: .utf8)
        var mixed = AgentTask(objective: "Explicit artifact action matrix", status: .failed, failure: "Later step failed")
        mixed.artifactRecords = [AgentArtifact("https://example.invalid/owned", kind: "webPage", title: "Owned page"),
            AgentArtifact(file.path, kind: "filePathRead", title: "Matrix file"),
            AgentArtifact(file.absoluteString, kind: "fileURLRead", title: "Matrix file URL"),
            AgentArtifact("owned-document-id", kind: "documentIDCreated", title: "Owned document"),
            AgentArtifact(task.durability!.receiptIDs.first!, kind: "receipt", title: "Action record"),
            AgentArtifact("https://", kind: "webPage"), AgentArtifact("javascript:alert(1)", kind: "webPage"),
            AgentArtifact("https://example.invalid/legacy")]
        let rows = AgentWorkingCard.artifactRows(for: mixed)
        check(rows.map(\.ordinal) == Array(0..<mixed.artifactRecords.count)
            && rows.map(\.artifact.value) == mixed.artifacts,
            "artifact control: mixed row order/value changed")
        opened.removeAll(); copied.removeAll()
        for index in [0, 1, 2] {
            check(AgentWorkingCard.performArtifactAction(.open, task: mixed, row: rows[index],
                open: { opened.append($0); return true }, copy: { _ in false }),
                "artifact feature: explicit mixed named Open refused row\(index)")
        }
        check(opened.count == 3 && opened[1] == file && opened[2] == file,
            "artifact feature: mixed URL/path/fileURL opened wrong target or blindly opened multiple")
        for index in [3, 4, 5, 6, 7] {
            let count = opened.count
            check(!AgentWorkingCard.performArtifactAction(.open, task: mixed, row: rows[index],
                open: { opened.append($0); return true }, copy: { _ in false }) && opened.count == count,
                "artifact feature: ID/receipt/malformed/legacy unknown guessed an Open target row\(index)")
            check(AgentWorkingCard.performArtifactAction(.copy, task: mixed, row: rows[index],
                open: { _ in false }, copy: { copied.append($0); return true }) && copied.last == mixed.artifacts[index],
                "artifact control: mixed Copy lost exact unknown text row\(index)")
        }
        try FileManager.default.removeItem(at: file)
        let count = opened.count
        check(!AgentWorkingCard.performArtifactAction(.open, task: mixed, row: rows[1],
            open: { opened.append($0); return true }, copy: { _ in false }) && opened.count == count,
            "artifact control: deleted file Open reached native backing")
        check(AgentWorkingCard.performArtifactAction(.copy, task: mixed, row: rows[1],
            open: { _ in false }, copy: { copied.append($0); return true }) && copied.last == file.path,
            "artifact control: deleted file lost copyable location")

        // Exact string-era snapshot, including whitespace/duplicates. No new record inference.
        let oldJSON = #"{"id":"legacy-artifact","objective":"Legacy fixture","source":"user","createdAt":"2023-11-14T22:13:20Z","contextReferences":[],"status":"completed","progress":"","artifacts":["  old/path  ","https://example.invalid/old","https://example.invalid/old","doc-id"],"arguments":{},"backend":"local","acpCLI":""}"#
        let encoder = JSONEncoder(); encoder.dateEncodingStrategy = .iso8601; encoder.outputFormatting = [.sortedKeys]
        let decoder = JSONDecoder(); decoder.dateDecodingStrategy = .iso8601
        let oldBytes = Data(oldJSON.utf8)
        let legacy = try decoder.decode(AgentTask.self, from: oldBytes)
        let expectedBytes = try JSONSerialization.data(withJSONObject: JSONSerialization.jsonObject(with: oldBytes),
            options: [.sortedKeys])
        let bytes = try encoder.encode(legacy)
        check(try decoder.decode(AgentTask.self, from: bytes) == legacy && bytes == expectedBytes
            && (JSONSerialization.jsonObject(with: bytes) as? [String: Any])?["artifactRecords"] == nil,
            "artifact control: legacy bytes/order/duplicates gained typed metadata")
        check(try store.save([legacy]).primaryCommitted && sql.load() == [legacy]
            && witnessTasks(directory.appendingPathComponent("agent-tasks.json")) == [legacy],
            "artifact control: decoded actual legacy SQL/witness canonical roundtrip changed")
        var invalid = try JSONSerialization.jsonObject(with: encoder.encode(mixed)) as! [String: Any]
        invalid["artifactRecords"] = [["value": "wrong", "kind": "documentCreated"]]
        var rejected = false
        do { _ = try decoder.decode(AgentTask.self, from: JSONSerialization.data(withJSONObject: invalid)) }
        catch { rejected = true }
        check(rejected, "artifact control: conflicting typed/legacy canonical paths were silently admitted")
        // Actual reader must retain historical fields, preserve canonical bytes, and refuse unsupported versions.
        try sql.withConnection { db in
            try TaskStore.exec(db, "UPDATE task_artifact SET kind='legacy-fixture-kind',title='Legacy retained title' WHERE ordinal=0")
        }
        check(try sql.load() == [legacy] && store.save([legacy]).primaryCommitted
            && sql.withConnection { try TaskStore.stringColumn($0, "SELECT title FROM task_artifact WHERE ordinal=0") } == ["Legacy retained title"],
            "artifact control: historical SQL kind/title became canonical or was erased on same path")
        for unsupported in ["artifact-v1:", "artifact-v2:documentCreated"] {
            try sql.withConnection { db in
                try TaskStore.run(db, "UPDATE task_artifact SET kind=? WHERE ordinal=0", [.text(unsupported)])
            }
            var refused = false
            do { _ = try sql.load() } catch { refused = true }
            check(refused, "artifact feature: malformed/future versioned SQL metadata was silently admitted")
        }
        try sql.withConnection { try TaskStore.exec($0, "UPDATE task_artifact SET kind='artifact-v1:futureMeaning',title='Unknown meaning' WHERE ordinal=0") }
        let unknown = try sql.load().first!
        check(unknown.artifactRecords.first?.kind == "futureMeaning"
            && AgentWorkingCard.artifactRows(for: unknown).first?.openURL == nil
            && AgentActivityProjector.artifactHeading(for: unknown) == "Results",
            "artifact feature: unknown versioned semantic authorized Open/created output")
        let duplicate = AgentToolResult(summary: "Same explicit legacy ref/link", reference: "https://example.invalid/same",
            link: URL(string: "https://example.invalid/same"))
        let id = UUID().uuidString
        AgentArtifactLedger.capture(taskID: id, result: duplicate)
        AgentArtifactLedger.capture(taskID: id, result: duplicate)
        check(AgentArtifactLedger.take(taskID: id) == ["https://example.invalid/same"]
            && AgentArtifactLedger.take(taskID: id).isEmpty,
            "artifact control: duplicate ref/link capture changed ordered one-consume semantics")
    }
