import Foundation

/// Unapplied: real runtime + isolated actual disk receipts, no task recovery claim.
@MainActor
enum ReceiptPersistenceRuntimeSelfTest {
    private enum FixtureError: Error { case invalidFixture }

    static func run() async -> String {
        guard SelfTest.isRunning else { return "RECEIPT_RUNTIME_FAILED: harness required" }
        let before = SelfTestStoreGuard.take()
        let root = FileManager.default.temporaryDirectory
            .appendingPathComponent("NextNotesReceiptRuntime-\(UUID().uuidString)", isDirectory: true)
        defer { try? FileManager.default.removeItem(at: root) }
        var failures: [String] = []
        var cases = 0
        func check(_ value: Bool, _ message: String) {
            cases += 1
            if !value { failures.append(message) }
        }
        do {
            try FileManager.default.createDirectory(at: root, withIntermediateDirectories: false)
            let file = root.appendingPathComponent("original.txt")
            let content = "Actual isolated receipt runtime read"
            try content.write(to: file, atomically: true, encoding: .utf8)
            guard let tool = AgentToolRegistry.shared.tool(named: "filesystem.read") else {
                throw FixtureError.invalidFixture
            }
            let arguments = ["path": file.path]
            let receiptFile = root.appendingPathComponent("success-receipts.json")
            let store = ActionReceiptStore(fileURL: receiptFile)
            let intent = ActionIntent(source: .agent, authority: .user, verb: tool.id,
                arguments: arguments, risk: tool.risk)
            var effects = 0
            var preEffectFired = false
            let result = try await ActionOrchestrator.shared.execute(
                intent: intent, tool: tool, title: "Read the isolated fixture",
                routing: ActionRouting(taskID: "isolated-receipt-success"),
                policy: .selfTest, promptIfNeeded: false, receiptStore: store,
                fire: { prepared in
                    guard prepared.executionPlan.toolID == tool.id,
                          prepared.executionPlan.arguments == arguments else { throw FixtureError.invalidFixture }
                    // A fresh disk reader must observe the exact .fired fact before
                    // the backing runs; the in-memory array cannot satisfy this.
                    let fresh = ActionReceiptStore(fileURL: receiptFile)
                    preEffectFired = fresh.receipt(forIntent: intent.id)?.status == .fired
                        && fresh.receipt(forIntent: intent.id)?.taskID == "isolated-receipt-success"
                        && fresh.receipt(forIntent: intent.id)?.preparedAction?.executionPlan.arguments == arguments
                    guard preEffectFired else { throw FixtureError.invalidFixture }
                    effects += 1
                    return try FilesystemExecutor.run(tool, arguments: arguments)
                })
            check(preEffectFired && effects == 1 && result.summary.contains(content),
                "runtime fired without a fresh readable exact .fired receipt or lost the original read")
            let completed = ActionReceiptStore(fileURL: receiptFile).receipt(forIntent: intent.id)
            check(completed?.status == .completed && completed?.events.contains(where: { $0.stage == .fired }) == true
                && completed?.taskID == "isolated-receipt-success" && completed?.result.contains(content) == true,
                "actual completed receipt did not survive a fresh disk reader")

            let rejectedFile = root.appendingPathComponent("rejected-receipts.json")
            let retainedFile = root.appendingPathComponent("retained-approved.json")
            let rejectedStore = ActionReceiptStore(fileURL: rejectedFile)
            let rejectedIntent = ActionIntent(source: .agent, authority: .user, verb: tool.id,
                arguments: arguments, risk: tool.risk)
            var exactDenied = false
            var faultReached = false
            var faultError = false
            var retainedBytes: Data?
            do {
                _ = try await ActionOrchestrator.shared.execute(
                    intent: rejectedIntent, tool: tool, title: "Refuse an unrecorded fixture effect",
                    routing: ActionRouting(taskID: "isolated-receipt-rejected"),
                    policy: .selfTest, promptIfNeeded: false,
                    isStillValid: {
                        // Inject failure only at the existing pre-fire boundary,
                        // after genuine approved persistence. No effect is replaced.
                        guard ActionReceiptStore(fileURL: rejectedFile).receipt(forIntent: rejectedIntent.id)?.status == .approved else {
                            return false
                        }
                        do {
                            retainedBytes = try Data(contentsOf: rejectedFile)
                            try FileManager.default.moveItem(at: rejectedFile, to: retainedFile)
                            try FileManager.default.createDirectory(at: rejectedFile, withIntermediateDirectories: false)
                            faultReached = true
                            return true
                        } catch { faultError = true; return false }
                    }, receiptStore: rejectedStore,
                    fire: { _ in
                        effects += 1
                        throw FixtureError.invalidFixture
                    })
            } catch let error as AgentError {
                if case .backendUnavailable(let message) = error {
                    exactDenied = message == "Could not save action execution; it was not run."
                }
            } catch { }
            check(faultReached && !faultError && exactDenied && effects == 1,
                "rejected .fired disk write did not refuse the exact pre-effect boundary")
            check(rejectedStore.receipt(forIntent: rejectedIntent.id)?.status == .approved
                && rejectedStore.receipt(forIntent: rejectedIntent.id)?.events.contains(where: { $0.stage == .fired }) == false,
                "a rejected disk write retained an uncommitted .fired receipt in memory")
            check((try? Data(contentsOf: retainedFile)) == retainedBytes
                && ActionReceiptStore(fileURL: retainedFile).receipt(forIntent: rejectedIntent.id)?.status == .approved
                && (try? FileManager.default.contentsOfDirectory(atPath: rejectedFile.path))?.isEmpty == true,
                "rejected execution changed retained approved bytes or wrote through its failing fixture target")
        } catch { failures.append("runtime fixture: \(error)") }
        failures += SelfTestStoreGuard.diff(before, SelfTestStoreGuard.take()).map { "owner changed: \($0)" }
        for failure in failures { SelfTest.diagnostic("RECEIPT_RUNTIME_WRONG: \(failure)") }
        return failures.isEmpty ? "RECEIPT_RUNTIME_OK: \(cases) cases"
            : "RECEIPT_RUNTIME_FAILED: \(failures.count) assertions"
    }
}
