types = '''/// A point-in-time result of the existing authority-bound primary row check.
/// Missing evidence is unknown; no health, effect or retry inference follows.
enum TaskWorkerPrimaryOutcome: String, Sendable { case accepted, mismatch, unavailable }

struct TaskWorkerPrimaryObservation: Equatable, Sendable {
    let expected: TaskWorkerWriteCondition
    let outcome: TaskWorkerPrimaryOutcome
    let observedAt: Date
    let state: AgentTaskStatus?
    let durability: TaskDurability?
}

'''
durability = sources['TaskDurability'].replace('/// Persisted optional execution metadata.', types + '/// Persisted optional execution metadata.', 1)

sql = sources['TaskStore'].replace('struct TaskWorkerWriteCondition: Sendable {', 'struct TaskWorkerWriteCondition: Equatable, Sendable {', 1)
(out / 'original-scaffold-TaskStore.swift.txt').write_text(sql)
sql = sql.replace('guard try Self.matchesWorker(worker, in: db),', 'guard try Self.workerPrimaryObservation(worker, in: db).outcome == .accepted,', 1)
start = sql.index('    /// Fail-fast existing-only primary read for the final worker validity hook.')
end = sql.index('    /// One transaction replaces', start)
sql = sql[:start] + '''    /// The Bool execution contract consumes the same single authoritative row read.
    func matchesWorker(_ worker: TaskWorkerWriteCondition, probe: TaskStoreAuthorityProbe) throws -> Bool {
        try workerPrimaryObservation(worker, probe: probe).outcome == .accepted
    }

    /// Return metadata only after existing(...) finishes its native file-binding check.
    func workerPrimaryObservation(_ worker: TaskWorkerWriteCondition,
                                  probe: TaskStoreAuthorityProbe) throws -> TaskWorkerPrimaryObservation {
        guard let authority = probe.authority else { throw TaskStoreError.authorityConflict }
        guard lock.try() else { throw TaskStoreError.sqlite(SQLITE_BUSY) }
        defer { lock.unlock() }
        return try existing(identity: probe.identity, writable: false) { db in
            try Self.exec(db, "BEGIN")
            defer { try? Self.exec(db, "ROLLBACK") }
            guard try Self.authority(in: db) == authority else { throw TaskStoreError.authorityConflict }
            return try Self.workerPrimaryObservation(worker, in: db)
        }
    }

    private static func workerPrimaryObservation(_ worker: TaskWorkerWriteCondition,
                                                 in db: OpaquePointer) throws -> TaskWorkerPrimaryObservation {
        let statement = try prepare(db, "SELECT state,durability FROM task WHERE id=?")
        defer { sqlite3_finalize(statement) }
        try bind(statement, [.text(worker.taskID)])
        let code = sqlite3_step(statement)
        guard code == SQLITE_ROW else {
            if code == SQLITE_DONE {
                return TaskWorkerPrimaryObservation(expected: worker, outcome: .mismatch,
                    observedAt: Date(), state: nil, durability: nil)
            }
            throw TaskStoreError.sqlite(code)
        }
        guard let state = try text(statement, 0), let status = AgentTaskStatus(rawValue: state) else {
            throw TaskStoreError.invalidRecord
        }
        let durability = try text(statement, 1).map { try decode(TaskDurability.self, $0) }
        guard (durability?.attempt ?? 0) >= 0 else { throw TaskStoreError.invalidRecord }
        let matches = status == worker.status && (durability?.attempt ?? 0) == worker.attempt
            && durability?.leaseOwner == worker.leaseOwner
        return TaskWorkerPrimaryObservation(expected: worker, outcome: matches ? .accepted : .mismatch,
            observedAt: Date(), state: status, durability: durability)
    }

''' + sql[end:]
(out / 'proposed-TaskStore.swift.txt').write_text(sql)

store = sources['AgentTaskStore']
start = store.index('    /// No creator/fallback path:')
end = store.index('    @discardableResult', start)
store = store[:start] + '''    /// Preserve the Bool execution contract and the same authority-bound row read.
    func matchesWorker(_ worker: TaskWorkerWriteCondition) -> Bool {
        workerPrimaryObservation(worker).outcome == .accepted
    }

    /// Any failed source/binding/read boundary returns unknown metadata, never absence.
    func workerPrimaryObservation(_ worker: TaskWorkerWriteCondition) -> TaskWorkerPrimaryObservation {
        let unavailable = TaskWorkerPrimaryObservation(expected: worker, outcome: .unavailable,
            observedAt: Date(), state: nil, durability: nil)
        guard !failedLoad else { return unavailable }
        guard !SelfTest.isRunning || allowsHarnessPersistence else { return unavailable }
        do {
            guard let probe = try mirror.probeAuthority(includeTasks: false) else { return unavailable }
            try check(probe, witness: try? compatibilityData())
            return try mirror.workerPrimaryObservation(worker, probe: probe)
        } catch { return unavailable }
    }

''' + store[end:]
(out / 'proposed-AgentTaskStore.swift.txt').write_text(store)

manager = sources['AgentTaskManager']
needle = '    /// Repair reuses the same startup hold transition;'
scaffold = '''    /// Last primary check attached to this exact live handle; performs no read.
    /// Unknown/mismatch metadata grants no health, effect or retry authority.
    func workerPrimaryObservation(taskID: String) -> TaskWorkerPrimaryObservation? { nil }

'''
(out / 'original-scaffold-AgentTaskManager.swift.txt').write_text(manager.replace(needle, scaffold + needle, 1))
manager = manager.replace('        var binding: TaskWorkerBinding? = nil\n', '        var binding: TaskWorkerBinding? = nil\n        var primaryObservation: TaskWorkerPrimaryObservation? = nil\n', 1)
getter = '''    /// Last primary check attached to this exact live handle; performs no read.
    /// Unknown/mismatch metadata grants no health, effect or retry authority.
    func workerPrimaryObservation(taskID: String) -> TaskWorkerPrimaryObservation? {
        guard let binding = workerBinding(taskID: taskID),
              let observed = running[taskID]?.primaryObservation,
              observed.expected.taskID == binding.taskID,
              observed.expected.attempt == binding.attempt,
              observed.expected.leaseOwner == binding.leaseOwner,
              observed.expected.status == .running else { return nil }
        return observed
    }

'''
manager = manager.replace(needle, getter + needle, 1)
old = '                return self.store.matchesWorker(expected)'
new = '''                let worker = self.running[taskID]
                let observed = self.store.workerPrimaryObservation(expected)
                // An obsolete callback may refuse work, but cannot stamp a successor.
                if let worker, !worker.handle.isCancelled, let binding = worker.binding,
                   binding.attempt == task.durability?.attempt,
                   binding.leaseOwner == task.durability?.leaseOwner,
                   binding.startedAt == task.durability?.attemptStartedAt,
                   let currentWorker = self.running[taskID], !currentWorker.handle.isCancelled,
                   currentWorker.binding == binding {
                    self.running[taskID]?.primaryObservation = observed
                    Log.app.debug("Task primary check: attempt \\(binding.attempt), outcome \\(observed.outcome.rawValue, privacy: .public)")
                }
                return observed.outcome == .accepted'''
assert manager.count(old) == 1
manager = manager.replace(old, new, 1)
(out / 'proposed-AgentTaskManager.swift.txt').write_text(manager)
