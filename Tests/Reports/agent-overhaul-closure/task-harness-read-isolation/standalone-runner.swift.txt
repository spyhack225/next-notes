

extension TaskHarnessReadIsolationSelfTest {
    /// Runs before App.main: the original task reader must not see owner UI stores.
    static func runStandalone() -> Int32 {
        let flag = "--selftest-task-harness-read"
        let arguments = Array(CommandLine.arguments.dropFirst())
        let allowed: Set<String> = [flag, SelfTest.outputFlag, SelfTest.timeoutFlag]
        guard SelfTest.requested == flag,
              arguments.filter({ $0.hasPrefix("--selftest") && $0 != SelfTest.outputFlag && $0 != SelfTest.timeoutFlag }) == [flag],
              arguments.filter({ $0.hasPrefix("-") }).allSatisfy({ allowed.contains($0) }) else {
            print("TASK_HARNESS_READ_FAILED: conflicting diagnostic modifier")
            return 1
        }
        if let output = SelfTest.outputPath {
            let temporary = FileManager.default.temporaryDirectory.resolvingSymlinksInPath().standardizedFileURL
            let candidate = URL(fileURLWithPath: output).standardizedFileURL.resolvingSymlinksInPath()
            guard candidate.path.hasPrefix(temporary.path + "/"),
                  !candidate.pathComponents.contains("Application Support") else {
                print("TASK_HARNESS_READ_FAILED: output must be a temporary diagnostic file")
                return 1
            }
        }
        let root = FileManager.default.temporaryDirectory
            .appendingPathComponent("NextNotesTaskRead-" + UUID().uuidString, isDirectory: true)
        defer { try? FileManager.default.removeItem(at: root) }
        var failures: [String] = []
        var cases = 0
        do {
            try FileManager.default.createDirectory(at: root, withIntermediateDirectories: false)
            try originalCases(root: root) { condition, message in
                cases += 1
                if !condition { failures.append(message) }
            }
        } catch { failures.append("isolated synthetic task history fixture could not complete") }
        for failure in failures { SelfTest.diagnostic("TASK_HARNESS_READ_WRONG: " + failure) }
        SelfTest.diagnostic(failures.isEmpty ? "TASK_HARNESS_READ_OK: \(cases) cases"
            : "TASK_HARNESS_READ_FAILED: \(failures.count) assertions across \(cases) cases")
        return failures.isEmpty ? 0 : 1
    }
}
