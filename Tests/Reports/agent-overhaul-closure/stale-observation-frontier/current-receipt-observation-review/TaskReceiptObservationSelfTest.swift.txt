import Foundation

/// Identical original nil-scaffold/fixed cases through actual Manager/runtime/read receipts.
/// CURRENT source metadata only; cached source quality and write admission stay unchanged.
@MainActor
enum TaskReceiptObservationSelfTest {
    private enum FixtureError: Error { case unsafeRoot, invalidBacking, missingReceipt, timeout }

    private static func settled(_ manager: AgentTaskManager, id: String) async -> Bool {
        let deadline = ContinuousClock.now.advanced(by: .seconds(5))
        while ContinuousClock.now < deadline {
            if let state = manager.task(id: id)?.status, state != .queued && state != .running { return true }
            await Task.yield()
        }
        return false
    }

    static func originalCases(root: URL, input: URL, text: String,
                              check: (Bool, String) -> Void) async throws {
        let safe = root.resolvingSymlinksInPath().standardizedFileURL
        let temporary = FileManager.default.temporaryDirectory.resolvingSymlinksInPath().standardizedFileURL
        let safeInput = input.resolvingSymlinksInPath().standardizedFileURL
        guard SelfTest.isRunning, safe.isFileURL, safe.path.hasPrefix(temporary.path + "/"),
              !safe.pathComponents.contains("Application Support"), safeInput.isFileURL,
              safeInput.path.hasPrefix(safe.path + "/") else { throw FixtureError.unsafeRoot }
        let ownerBefore = SelfTestStoreGuard.take()
        let oldFire = AgentToolExecutor.fireOverrideForTesting
        let oldFake = AgentToolExecutor.fakeForTesting
        let oldPolicy = AgentToolExecutor.policyOverrideForTesting
        defer {
            AgentToolExecutor.fireOverrideForTesting = oldFire
            AgentToolExecutor.fakeForTesting = oldFake
            AgentToolExecutor.policyOverrideForTesting = oldPolicy
        }
        AgentToolExecutor.fakeForTesting = nil
        AgentToolExecutor.policyOverrideForTesting = .selfTest
        for mode in ["accepted", "source-changed"] {
            AgentToolExecutor.fireOverrideForTesting = nil
            let directory = safe.appendingPathComponent("receipt-current-" + mode, isDirectory: true)
            try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: false)
            let sql = TaskStore(root: directory)
            defer { sql.close() }
            let export = directory.appendingPathComponent("agent-tasks.json")
            let receiptFile = directory.appendingPathComponent("action-receipts.json")
            let primerReceipts = ActionReceiptStore(fileURL: receiptFile)
            let primerStore = AgentTaskStore(fileURL: export, mirror: sql)
            guard primerStore.allowsHarnessPersistence else { throw FixtureError.unsafeRoot }
            let primer = AgentTaskManager(store: primerStore, receiptStore: primerReceipts)
            let primerTask = primer.submit(objective: "Create actual current-receipt source", tool: "filesystem.read",
                arguments: ["path": input.path], source: "selftest")
            guard await settled(primer, id: primerTask.id) else {
                primer.cancel(primerTask.id); throw FixtureError.timeout
            }
            check(primer.task(id: primerTask.id)?.status == .completed
                && primer.task(id: primerTask.id)?.result?.contains(text) == true
                && primerReceipts.receipts.count == 1 && primerReceipts.receipts.first?.status == .completed,
                "current receipt: real primer control failed (\(mode))")
            guard let primerReceipt = primerReceipts.receipts.first else { throw FixtureError.missingReceipt }
            let receipts = ActionReceiptStore(fileURL: receiptFile)
            let initialQuality = receipts.sourceObservation
            let store = AgentTaskStore(fileURL: export, mirror: sql)
            guard store.allowsHarnessPersistence else { throw FixtureError.unsafeRoot }
            let manager = AgentTaskManager(store: store, receiptStore: receipts)
            var context: TaskJournalContext?
            var effects = 0
            var announcements: [String] = []
            var fired: ActionReceipt?
            var atFire: ActionReceiptCurrentSourceObservation?
            var afterExternalChange: ActionReceiptCurrentSourceObservation?
            var filtered: [ActionReceiptCurrentSourceObservation?] = []
            var invalid: [ActionReceiptCurrentSourceObservation?] = []
            var exactJoin = false
            var probeInvariant = false
            var changedBytes: Data?
            var qualityAtFire: ActionReceiptSourceObservation?
            var qualityAfterExternalChange: ActionReceiptSourceObservation?
            manager.announcementForTesting = { announcements.append($0) }
            manager.backendEntryForTesting = { context = TaskEventJournal.current }
            AgentToolExecutor.fireOverrideForTesting = { tool, arguments in
                guard let context, let owner = context.leaseOwner, context.attempt > 0,
                      context.receiptStore === receipts, tool.id == "filesystem.read",
                      arguments == ["path": input.path] else { throw FixtureError.invalidBacking }
                let reader = ActionReceiptStore(fileURL: receiptFile)
                guard let saved = reader.receipts.first(where: { $0.taskID == context.taskID && $0.status == .fired })
                else { throw FixtureError.invalidBacking }
                fired = saved
                let primaryBefore = try sql.load()
                let journalBefore = try sql.journal(taskID: context.taskID)
                let exportBefore = try Data(contentsOf: export)
                let bytesBefore = try Data(contentsOf: receiptFile)
                let memoryBefore = receipts.receipts
                qualityAtFire = receipts.sourceObservation
                let row = primaryBefore.first { $0.id == context.taskID }
                let ids = row?.durability?.receiptIDs ?? []
                exactJoin = row?.status == .running && ids == [saved.actionID.uuidString]
                    && saved.taskAttempt == context.attempt && saved.taskLeaseOwner == owner
                    && owner == String(ProcessInfo.processInfo.processIdentifier)
                    && saved.preparedAction?.id == saved.actionID
                    && saved.preparedAction?.executionPlan.arguments == arguments
                atFire = receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt,
                    leaseOwner: owner, receiptIDs: ids + ["malformed-retained-link", primerReceipt.actionID.uuidString])
                filtered = [
                    receipts.currentSourceObservation(taskID: "other-task", attempt: context.attempt,
                        leaseOwner: owner, receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt + 1,
                        leaseOwner: owner, receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt,
                        leaseOwner: "other-owner", receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt,
                        leaseOwner: owner, receiptIDs: ["malformed-retained-link"])
                ]
                invalid = [
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: 0,
                        leaseOwner: owner, receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: -1,
                        leaseOwner: owner, receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt,
                        leaseOwner: " \n", receiptIDs: ids),
                    receipts.currentSourceObservation(taskID: "", attempt: context.attempt,
                        leaseOwner: owner, receiptIDs: ids)
                ]
                for _ in 0..<3 {
                    _ = receipts.currentSourceObservation(taskID: context.taskID, attempt: context.attempt,
                        leaseOwner: owner, receiptIDs: ids)
                }
                let primaryAfter = try sql.load()
                let journalAfter = try sql.journal(taskID: context.taskID)
                let exportAfter = try Data(contentsOf: export)
                let bytesAfter = try Data(contentsOf: receiptFile)
                probeInvariant = primaryAfter == primaryBefore && journalAfter == journalBefore
                    && exportAfter == exportBefore && bytesAfter == bytesBefore
                    && receipts.receipts == memoryBefore && receipts.sourceObservation == qualityAtFire
                // The same original read remains the actual backing.
                let result = try FilesystemExecutor.run(tool, arguments: arguments)
                effects += 1
                if mode == "source-changed" {
                    guard var rows = try JSONSerialization.jsonObject(with: bytesBefore) as? [[String: Any]],
                          rows.count == 2 else { throw FixtureError.invalidBacking }
                    rows.append(["unreadable_future_receipt": true])
                    let data = try JSONSerialization.data(withJSONObject: rows, options: [.sortedKeys])
                    try data.write(to: receiptFile, options: .atomic)
                    changedBytes = data
                }
                afterExternalChange = receipts.currentSourceObservation(taskID: context.taskID,
                    attempt: context.attempt, leaseOwner: owner, receiptIDs: ids)
                qualityAfterExternalChange = receipts.sourceObservation
                return result
            }
            let submitted = manager.submit(objective: "Read current-receipt fixture", tool: "filesystem.read",
                arguments: ["path": input.path], source: "selftest")
            guard await settled(manager, id: submitted.id) else {
                manager.cancel(submitted.id); throw FixtureError.timeout
            }
            guard let fired, let attempt = fired.taskAttempt, let owner = fired.taskLeaseOwner else {
                throw FixtureError.missingReceipt
            }
            let ids = [fired.actionID.uuidString]
            let final = manager.task(id: submitted.id)
            let primary = try sql.load().first { $0.id == submitted.id }
            let events = try sql.journal(taskID: submitted.id)
            let terminal = receipts.currentSourceObservation(taskID: submitted.id, attempt: attempt,
                leaseOwner: owner, receiptIDs: ids)
            let fresh = ActionReceiptStore(fileURL: receiptFile)
            let freshCurrent = fresh.currentSourceObservation(taskID: submitted.id, attempt: attempt,
                leaseOwner: owner, receiptIDs: ids)
            check(effects == 1 && exactJoin && manager.backendStartsForTesting == 1,
                "current receipt: real target backing lost exact entry/join (\(mode))")
            check(initialQuality?.loadQuality == .readable && initialQuality?.loadedRowCount == 1
                && qualityAtFire?.lastWriteSourceCheck == .matched,
                "current receipt: existing historical readable/write control failed (\(mode))")
            check(probeInvariant && qualityAfterExternalChange == qualityAtFire,
                "current receipt: CURRENT probes changed primary/journal/export/receipt/cache (\(mode))")
            check(atFire?.state == .matchingSource && atFire?.retainedFingerprint != nil
                && atFire?.retainedFingerprint == atFire?.observedFingerprint,
                "current receipt: actual fired source lacks CURRENT matching fingerprint evidence (\(mode))")
            check(atFire?.receipts.count == 1 && atFire?.receipts.first?.actionID == fired.actionID
                && atFire?.receipts.first?.preparedActionID == fired.actionID
                && atFire?.receipts.first?.taskID == submitted.id && atFire?.receipts.first?.taskAttempt == attempt
                && atFire?.receipts.first?.taskLeaseOwner == owner && atFire?.receipts.first?.status == .fired
                && atFire?.receipts.first?.startedAt == fired.startedAt
                && atFire?.receipts.first?.eventStages.contains(.prepared) == true
                && atFire?.receipts.first?.eventStages.contains(.fired) == true,
                "current receipt: actual prepared/fired metadata or exact filter missing (\(mode))")
            check(atFire?.sourceObservation == qualityAtFire,
                "current receipt: CURRENT comparison concealed separate historical source quality (\(mode))")
            for (index, observation) in filtered.enumerated() {
                check(observation?.state == .matchingSource && observation?.receipts.isEmpty == true,
                    "current receipt: wrong binding/malformed-only links obtained cached rows filter\(index) (\(mode))")
            }
            for (index, observation) in invalid.enumerated() {
                check(observation?.state == .unavailable && observation?.observedFingerprint == nil
                    && observation?.receipts.isEmpty == true,
                    "current receipt: invalid expected binding was observed or admitted invalid\(index) (\(mode))")
            }
            check(primary?.durability?.receiptIDs == ids && final?.durability?.receiptIDs == ids,
                "current receipt: terminal history lost original exact link (\(mode))")
            if mode == "accepted" {
                check(afterExternalChange?.state == .matchingSource && afterExternalChange == atFire,
                    "current receipt: unchanged source probe did not retain exact fired observation")
                check(final?.status == .completed && primary?.status == .completed
                    && final?.result?.contains(text) == true && fresh.receipt(for: fired.actionID)?.status == .completed
                    && announcements.contains { $0.contains(text) } && events.contains { $0.draft.kind == .jobCompleted },
                    "current receipt: actual accepted read/result delivery changed")
                check(terminal?.state == .matchingSource && terminal?.receipts.first?.status == .completed
                    && terminal?.receipts.first?.eventStages.contains(.completed) == true
                    && terminal?.retainedFingerprint == terminal?.observedFingerprint
                    && terminal?.sourceObservation?.loadedRowCount == 1,
                    "current receipt: actual own terminal write did not refresh current source while retaining load history")
                check(freshCurrent?.state == .matchingSource && freshCurrent?.sourceObservation?.loadQuality == .readable
                    && freshCurrent?.sourceObservation?.loadedRowCount == 2
                    && freshCurrent?.receipts.first?.status == .completed,
                    "current receipt: actual fresh completed reader lacks matching source/bound receipt")
                try boundaryCases(directory: directory, file: receiptFile, store: receipts,
                    taskID: submitted.id, attempt: attempt, owner: owner, ids: ids, check: check)
            } else {
                let bytes = try Data(contentsOf: receiptFile)
                check(afterExternalChange?.state == .changed && afterExternalChange?.receipts.isEmpty == true
                    && afterExternalChange?.retainedFingerprint == atFire?.retainedFingerprint
                    && afterExternalChange?.observedFingerprint != afterExternalChange?.retainedFingerprint,
                    "current receipt: source changed before terminal save remained CURRENT matched/cached positive")
                check(terminal?.state == .changed && terminal?.receipts.isEmpty == true
                    && terminal?.sourceObservation?.lastWriteSourceCheck == .mismatch,
                    "current receipt: refused terminal source concealed changed CURRENT vs LAST write evidence")
                check(bytes == changedBytes && receipts.receipt(for: fired.actionID) == fired
                    && fresh.receipt(for: fired.actionID) == fired,
                    "current receipt: source refusal/probes erased exact changed bytes or real fired evidence")
                check(freshCurrent?.state == .matchingSource && freshCurrent?.sourceObservation?.loadQuality == .partial
                    && freshCurrent?.sourceObservation?.loadedRowCount == 2
                    && freshCurrent?.sourceObservation?.unreadableRowCount == 1
                    && freshCurrent?.receipts.first?.status == .fired,
                    "current receipt: current partial reader hid readable fired metadata or claimed complete quality")
                let failure = "The action ran, but its result could not be saved. Check it before trying again."
                check(final?.status == .failed && primary?.status == .failed && final?.result == nil
                    && final?.failure == failure && primary?.failure == failure && announcements.contains(failure)
                    && !announcements.contains { $0.contains(text) }
                    && !events.contains { $0.draft.kind == .jobCompleted },
                    "current receipt: source observation changed original result-not-saved/no-success flow")
            }
            SelfTest.diagnostic("TASK_RECEIPT_CURRENT_TRACE: mode=\(mode) effects=\(effects) terminal=\(final?.status.rawValue ?? "absent") current=\(terminal?.state.rawValue ?? "unobserved") freshQuality=\(freshCurrent?.sourceObservation?.loadQuality.rawValue ?? "unobserved")")
        }
        check(SelfTestStoreGuard.diff(ownerBefore, SelfTestStoreGuard.take()).isEmpty,
            "current receipt: isolated actual runtime changed owner stores/defaults")
    }

    private static func boundaryCases(directory: URL, file: URL, store: ActionReceiptStore,
                                      taskID: String, attempt: Int, owner: String, ids: [String],
                                      check: (Bool, String) -> Void) throws {
        let memory = store.receipts
        let quality = store.sourceObservation
        let original = try Data(contentsOf: file)
        let current = store.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        let handle = try FileHandle(forWritingTo: file)
        try handle.seekToEnd()
        try handle.write(contentsOf: Data([0x20]))
        try handle.close()
        let edited = store.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(edited?.state == .changed && edited?.receipts.isEmpty == true
            && edited?.retainedFingerprint == current?.retainedFingerprint
            && edited?.observedFingerprint != current?.observedFingerprint,
            "current receipt: same-file metadata edit remained matched")
        let retained = directory.appendingPathComponent("retained-real-receipts.json")
        try FileManager.default.moveItem(at: file, to: retained)
        try original.write(to: file, options: .atomic)
        let replaced = store.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(replaced?.state == .changed && replaced?.receipts.isEmpty == true,
            "current receipt: identical-content replacement erased file-identity evidence")
        let replacementBytes = try Data(contentsOf: file)
        check(replacementBytes == original && store.receipts == memory && store.sourceObservation == quality,
            "current receipt: replacement probe changed exact source/cache")
        try FileManager.default.removeItem(at: file)
        let deleted = store.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(deleted?.state == .changed && deleted?.observedFingerprint == .missing
            && deleted?.receipts.isEmpty == true && !FileManager.default.fileExists(atPath: file.path),
            "current receipt: deleted previously admitted source recreated or retained current rows")
        try FileManager.default.createDirectory(at: file, withIntermediateDirectories: false)
        let nonregular = store.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(nonregular?.state == .unavailable && nonregular?.observedFingerprint == nil
            && nonregular?.receipts.isEmpty == true && store.receipts == memory && store.sourceObservation == quality,
            "current receipt: nonregular source claimed metadata match or changed cache")
        let missingFile = directory.appendingPathComponent("never-created-receipts.json")
        let missing = ActionReceiptStore(fileURL: missingFile)
        let missingCurrent = missing.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(missingCurrent?.state == .matchingSource && missingCurrent?.retainedFingerprint == .missing
            && missingCurrent?.observedFingerprint == .missing && missingCurrent?.sourceObservation?.loadQuality == .missing
            && missingCurrent?.receipts.isEmpty == true && !FileManager.default.fileExists(atPath: missingFile.path),
            "current receipt: unchanged missing source lost explicit missing quality or created evidence")
        let unreadableFile = directory.appendingPathComponent("unreadable-receipts.json")
        let unreadableBytes = Data("unreadable receipt array".utf8)
        try unreadableBytes.write(to: unreadableFile, options: .atomic)
        let unreadable = ActionReceiptStore(fileURL: unreadableFile)
        let unreadableCurrent = unreadable.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        let unreadableAfter = try Data(contentsOf: unreadableFile)
        check(unreadableCurrent?.state == .unavailable && unreadableCurrent?.retainedFingerprint == nil
            && unreadableCurrent?.observedFingerprint == nil && unreadableCurrent?.sourceObservation?.loadQuality == .unreadable
            && unreadableCurrent?.receipts.isEmpty == true && unreadableAfter == unreadableBytes,
            "current receipt: undecodable historical source became current empty/full evidence")
        let refusedFile = directory.appendingPathComponent("Application Support", isDirectory: true)
            .appendingPathComponent("refused-receipts.json")
        let refused = ActionReceiptStore(fileURL: refusedFile)
        let refusedCurrent = refused.currentSourceObservation(taskID: taskID, attempt: attempt, leaseOwner: owner, receiptIDs: ids)
        check(refusedCurrent?.state == .unavailable && refusedCurrent?.observedFingerprint == nil
            && refusedCurrent?.sourceObservation?.loadQuality == .harnessRefused
            && refusedCurrent?.receipts.isEmpty == true && !FileManager.default.fileExists(atPath: refusedFile.path),
            "current receipt: harness-refused path obtained current source or created store")
    }
}
