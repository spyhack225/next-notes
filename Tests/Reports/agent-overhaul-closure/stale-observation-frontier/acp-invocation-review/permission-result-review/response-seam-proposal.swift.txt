// Exact precedent: existing request(method:paramsJSON:timeout:) already accepts
// actor-encoded Sendable Data and deserializes it inside this transport.
func reply(id: Any, resultJSON: Data) throws {
    guard let result = try JSONSerialization.jsonObject(with: resultJSON) as? [String: Any] else {
        throw JSONRPCError(message: "JSON-RPC result must be a JSON object.")
    }
    try write(["jsonrpc": "2.0", "id": id, "result": result])
}
// dispatch's existing Task passes the returned Data to reply(id:resultJSON:).
// Keep same existing state lock, pending table, notification ordering and EOF path.

// Session response encoding stays local and Foundation based. Selection uses
// actual offered IDs (including proper JSON escaping), never JSON interpolation.
private static let cancelledPermissionResult = Data(
    #"{"outcome":{"outcome":"cancelled"}}"#.utf8)
private static func selectedPermissionResult(_ optionID: String) -> Data {
    (try? JSONSerialization.data(withJSONObject:
        ["outcome": ["outcome": "selected", "optionId": optionID]]))
        ?? cancelledPermissionResult // Internal encode failure never becomes allow.
}
