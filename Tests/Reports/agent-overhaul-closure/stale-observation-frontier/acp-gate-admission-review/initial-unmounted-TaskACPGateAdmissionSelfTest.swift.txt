import Foundation

/// Actual Manager -> ACPBackend -> Session -> owned peer, same original/fixed inputs.
/// Actual Gate request/decision admission; no nested effect is ever executed.
@MainActor
enum TaskACPGateAdmissionSelfTest {
    private enum FixtureError: Error {
        case unsafeRoot, busyGate, missingContext, missingRow, missingReceipt, timeout, invalidPeer
    }

    private static func reached(_ condition: @MainActor () -> Bool) async -> Bool {
        let deadline = ContinuousClock.now.advanced(by: .seconds(5))
        while ContinuousClock.now < deadline {
            if condition() { return true }
            await Task.yield()
        }
        return condition()
    }

    // Match the existing AgentTaskStore canonical ISO export before its SQL save.
    // Live handle identity keeps the exact original in-memory producer timestamp.
    private static func canonicalDate(_ date: Date) throws -> Date {
        let encoder = JSONEncoder()
        encoder.dateEncodingStrategy = .iso8601
        let decoder = JSONDecoder()
        decoder.dateDecodingStrategy = .iso8601
        return try decoder.decode(Date.self, from: encoder.encode(date))
    }

    private static func object(_ url: URL) -> [String: Any] {
        guard let data = try? Data(contentsOf: url),
              let object = try? JSONSerialization.jsonObject(with: data) as? [String: Any] else { return [:] }
        return object
    }

    static func originalCases(root: URL, check: (Bool, String) -> Void) async throws {
        let safe = root.resolvingSymlinksInPath().standardizedFileURL
        let temporary = FileManager.default.temporaryDirectory.resolvingSymlinksInPath().standardizedFileURL
        guard SelfTest.isRunning, safe.isFileURL, safe.path.hasPrefix(temporary.path + "/"),
              !safe.pathComponents.contains("Application Support") else { throw FixtureError.unsafeRoot }
        guard PermissionGate.shared.pending == nil, PermissionGate.shared.queuedCount == 0 else {
            throw FixtureError.busyGate
        }
        let ownerBefore = SelfTestStoreGuard.take()
        let grantsBefore = PermissionGrantStore.shared.grants
        let priorFake = AgentToolExecutor.fakeForTesting
        let priorFire = AgentToolExecutor.fireOverrideForTesting
        defer {
            AgentToolExecutor.fakeForTesting = priorFake
            AgentToolExecutor.fireOverrideForTesting = priorFire
        }
        AgentToolExecutor.fakeForTesting = nil
        var unexpectedTools = 0
        AgentToolExecutor.fireOverrideForTesting = { _, _ in
            unexpectedTools += 1
            throw FixtureError.invalidPeer
        }
        let owner = String(ProcessInfo.processInfo.processIdentifier)
        let objective = "Owned ACP nested review fixture: no tools or external effects"
        let marker = "Owned ACP nested fixture finished."
        let sessionID = "fixture-acp-nested-attempt"
        let warning = "\nNext Notes could not independently verify the effect. Inspect the target before retrying."

        for mode in ["accepted", "request-rejected", "approval-rejected", "denial-rejected", "fresh-primary-held", "cancel-denial-rejected"] {
            let heldMode = mode == "fresh-primary-held"
            let requestRejected = mode == "request-rejected"
            let approvalRejected = mode == "approval-rejected"
            let denialRejected = mode == "denial-rejected"
            let cancelRejected = mode == "cancel-denial-rejected"
            let decisionRejected = approvalRejected || denialRejected
            let directory = safe.appendingPathComponent("acp-gate-admission-" + mode, isDirectory: true)
            try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: false)
            let sql = TaskStore(root: directory)
            defer { sql.close() }
            let export = directory.appendingPathComponent("agent-tasks.json")
            let store = AgentTaskStore(fileURL: export, mirror: sql)
            guard store.allowsHarnessPersistence else { throw FixtureError.unsafeRoot }
            let receiptFile = directory.appendingPathComponent("action-receipts.json")
            let script = directory.appendingPathComponent("owned-peer.py")
            let configFile = directory.appendingPathComponent("config.json")
            let releaseFile = directory.appendingPathComponent("release-permission.json")
            let wireFile = directory.appendingPathComponent("wire-result.json")
            let callsFile = directory.appendingPathComponent("calls.json")
            try peer.write(to: script, atomically: true, encoding: .utf8)
            let receipts = ActionReceiptStore(fileURL: receiptFile)
            let manager = AgentTaskManager(store: store, receiptStore: receipts)
            var context: TaskJournalContext?
            var rowAtEntry: AgentTask?
            var bindingAtEntry: TaskWorkerBinding?
            var entryValid = false
            var entered = 0
            var ended = 0
            var announcements: [String] = []
            var holdingSQL: TaskStore?
            var holdingManager: AgentTaskManager?
            defer { holdingSQL?.close() }
            let askBefore = PermissionGate.shared.askCount
            let toolsBefore = unexpectedTools
            manager.announcementForTesting = { announcements.append($0) }
            manager.backendExitForTesting = { ended += 1 }
            manager.backendEntryForTesting = {
                entered += 1
                guard let captured = TaskEventJournal.current else { throw FixtureError.missingContext }
                context = captured
                let reader = TaskStore(root: directory)
                defer { reader.close() }
                guard let row = try reader.load().first(where: { $0.id == captured.taskID }) else {
                    throw FixtureError.missingRow
                }
                rowAtEntry = row
                bindingAtEntry = manager.workerBinding(taskID: captured.taskID)
                entryValid = captured.isStillValid?() == true
                let config: [String: Any] = ["marker": marker, "objective": objective,
                    "taskID": captured.taskID, "taskAttempt": captured.attempt,
                    "taskLeaseOwner": captured.leaseOwner ?? "", "database": sql.fileURL.path,
                    "receipts": receiptFile.path]
                try JSONSerialization.data(withJSONObject: config, options: [.sortedKeys])
                    .write(to: configFile, options: .atomic)
            }
            // Common original/fixed SelfTest-only switch reaches actual Broker and Gate.
            let submitted = manager.submit(objective: objective, arguments: ["acpFixture": script.path, "acpFixtureAskPermission": "true"],
                backend: .acp, source: "selftest")
            do {
                guard await reached({ FileManager.default.fileExists(atPath:
                    directory.appendingPathComponent("prompt-observed.json").path) }) else { throw FixtureError.timeout }
                guard !FileManager.default.fileExists(atPath: directory.appendingPathComponent("peer-error.json").path),
                      let captured = context, let binding = bindingAtEntry else { throw FixtureError.invalidPeer }
                let startupReceipts = ActionReceiptStore(fileURL:
                    directory.appendingPathComponent("startup-fired-snapshot.json"))
                guard startupReceipts.receipts.count == 1, let fired = startupReceipts.receipts.first else {
                    throw FixtureError.missingReceipt
                }
                let joined = [fired.actionID.uuidString]
                let startup = object(directory.appendingPathComponent("startup-primary.json"))
                let startupDurability = startup["durability"] as? [String: Any] ?? [:]
                let startupEvents = startup["toolStartEvents"] as? [[String: Any]] ?? []
                let prompt = object(directory.appendingPathComponent("prompt-observed.json"))
                // C01
                check(entered == 1 && ended == 0 && manager.backendStartsForTesting == 1,
                    "ACP Gate admission: actual backend was not suspended in its first prompt (\(mode))")
                // C02
                check(rowAtEntry?.status == .running && rowAtEntry?.durability?.attempt == 1
                    && rowAtEntry?.durability?.leaseOwner == owner
                    && rowAtEntry?.durability?.runtimeClass == .acpWorker
                    && rowAtEntry?.durability?.resumePolicy == .neverAuto
                    && rowAtEntry?.durability?.attemptStartedAt != nil,
                    "ACP Gate admission: original backend lacked admitted normal typed ACP ownership (\(mode))")
                let canonicalBindingStart = try canonicalDate(binding.startedAt)
                SelfTest.diagnostic("TASK_ACP_NESTED_ENTRY_TRACE: mode=\(mode) entryValid=\(entryValid) receiptOwner=\(captured.receiptStore === receipts) memoryStart=\(binding.startedAt.timeIntervalSince1970) diskStart=\(rowAtEntry?.durability?.attemptStartedAt?.timeIntervalSince1970 ?? -1) canonicalStart=\(canonicalBindingStart.timeIntervalSince1970)")
                // C03
                check(captured.taskID == submitted.id && captured.attempt == 1 && captured.leaseOwner == owner
                    && captured.receiptStore === receipts && entryValid
                    && binding.taskID == submitted.id && binding.attempt == 1 && binding.leaseOwner == owner
                    && binding.startedAt == manager.task(id: submitted.id)?.durability?.attemptStartedAt
                    && canonicalBindingStart == rowAtEntry?.durability?.attemptStartedAt,
                    "ACP Gate admission: original live context lost exact owning manager/receipt/handle (\(mode))")
                // C04
                check(rowAtEntry?.durability?.receiptIDs.isEmpty == true,
                    "ACP Gate admission: receipt was joined before real preparation (\(mode))")
                // C05
                check(prompt["taskID"] as? String == submitted.id && prompt["sessionID"] as? String == sessionID
                    && prompt["promptCount"] as? Int == 1,
                    "ACP Gate admission: peer did not observe the real exact first prompt (\(mode))")
                // C06
                check(fired.taskID == submitted.id && fired.taskAttempt == 1 && fired.taskLeaseOwner == owner
                    && fired.status == .fired && fired.events.last?.stage == .fired
                    && fired.toolID == "mcp.acp_session" && fired.preparedAction?.id == fired.actionID
                    && fired.preparedAction?.routing.taskID == submitted.id
                    && fired.preparedAction?.intentID == fired.intentID
                    && fired.preparedAction?.executionPlan.toolID == fired.toolID
                    && fired.preparedAction?.executionPlan.arguments["objective"] == objective,
                    "ACP Gate admission: owned process lacked exact actual saved fired preparation (\(mode))")
                // C07
                check(startup["taskID"] as? String == submitted.id && startup["state"] as? String == "running"
                    && startup["actionID"] as? String == fired.actionID.uuidString
                    && startupDurability["attempt"] as? Int == 1
                    && startupDurability["leaseOwner"] as? String == owner
                    && startupDurability["receiptIDs"] as? [String] == joined
                    && startupEvents.count == 1 && startupEvents.first?["detail"] as? String == "mcp.acp_session"
                    && startupEvents.first?["attempt"] as? Int == 1,
                    "ACP Gate admission: actual startup missed exact primary receipt/tool-start join (\(mode))")
                // C08: workerBinding itself refuses a cancelled handle.
                check(manager.workerBinding(taskID: submitted.id) == binding
                    && !FileManager.default.fileExists(atPath: wireFile.path),
                    "ACP Gate admission: original handle was cancelled or permission ran before release (\(mode))")
                guard let releaseRow = try sql.load().first(where: { $0.id == submitted.id }) else {
                    throw FixtureError.missingRow
                }
                let releaseJournal = try sql.journal(taskID: submitted.id)
                if requestRejected {
                    try sql.withConnection { db in
                        try TaskStore.exec(db, "CREATE TRIGGER reject_gate_request BEFORE INSERT ON task_event WHEN NEW.kind='permissionRequested' BEGIN SELECT RAISE(ABORT,'owned request rejection'); END")
                    }
                }
                try JSONSerialization.data(withJSONObject: ["release": true], options: [.sortedKeys])
                    .write(to: releaseFile, options: .atomic)
                guard await reached({ PermissionGate.shared.pending?.taskID == submitted.id
                    || FileManager.default.fileExists(atPath: wireFile.path) }) else { throw FixtureError.timeout }
                let gate = PermissionGate.shared
                var requestID: String?
                var expectedApproved = mode == "accepted" || approvalRejected
                var expectedRequestCount = 1
                var expectedDecisionCount = 1
                var expectedDecisionKind: TaskEventKind = expectedApproved ? .permissionApproved : .permissionDenied
                var expectedFinalState: AgentTaskStatus = .completed
                var heldRow: AgentTask?
                var heldJournal: [TaskJournalEvent]?
                var cancelledForCleanup = false
                // C09
                check(gate.askCount == askBefore + 1 && gate.queuedCount == 0,
                    "ACP Gate admission: actual Session/Broker did not reach exactly one Gate ask (\(mode))")
                if requestRejected {
                    expectedApproved = false
                    expectedRequestCount = 0
                    expectedDecisionCount = 0
                    expectedDecisionKind = .permissionDenied
                    // R01: original wrongly presents a live card after rejected journal save.
                    check(gate.pending == nil && gate.pendingReview == nil,
                        "ACP Gate admission: rejected request still presented a live card")
                    let rejectedRow = try sql.load().first { $0.id == submitted.id }
                    let rejectedJournal = try sql.journal(taskID: submitted.id)
                    // R02
                    check(rejectedRow == releaseRow && rejectedJournal == releaseJournal
                        && manager.lastPersistenceResult?.primaryCommitted == false,
                        "ACP Gate admission: actual rejected request transaction changed primary history or was not rejected")
                    // Finish the OLD bad card honestly. This is a denied user response, not cancellation.
                    if let badCard = gate.pending, badCard.taskID == submitted.id {
                        requestID = badCard.id
                        _ = gate.respond(id: badCard.id, approved: false, duration: .once)
                    }
                    guard await reached({ FileManager.default.fileExists(atPath: wireFile.path) }) else { throw FixtureError.timeout }
                    try sql.withConnection { try TaskStore.exec($0, "DROP TRIGGER reject_gate_request") }
                    // R03
                    check(gate.pending == nil && gate.queuedCount == 0,
                        "ACP Gate admission: rejected request failed to release its actual waiter")
                } else {
                    guard let request = gate.pending, request.taskID == submitted.id,
                          let review = gate.pendingReview else { throw FixtureError.invalidPeer }
                    requestID = request.id
                    let reviewToolID = AgentToolRegistry.shared.tool(named: request.toolID)?.id ?? "mcp." + request.toolID
                    let requested = try sql.journal(taskID: submitted.id)
                    // C10 (nonrequest modes)
                    check(request.toolID == "mcp.acp_nested_tool" && request.arguments["taskID"] == submitted.id
                        && review.id == request.id && review.toolID == reviewToolID && review.isReadyToRun
                        && requested.filter { $0.draft.kind == .permissionRequested && $0.draft.attempt == 1 }.count == 1
                        && !FileManager.default.fileExists(atPath: wireFile.path),
                        "ACP Gate admission: actual pending review/request journal or suspended wire control failed (\(mode))")
                    if decisionRejected || cancelRejected {
                        let rejectedKind = approvalRejected ? "permissionApproved" : "permissionDenied"
                        try sql.withConnection { db in
                            try TaskStore.exec(db, "CREATE TRIGGER reject_gate_decision BEFORE INSERT ON task_event WHEN NEW.kind='\(rejectedKind)' BEGIN SELECT RAISE(ABORT,'owned decision rejection'); END")
                        }
                    }
                    if decisionRejected {
                        let beforeDecision = try sql.load().first { $0.id == submitted.id }
                        let beforeJournal = try sql.journal(taskID: submitted.id)
                        let admitted = gate.respond(id: request.id, approved: approvalRejected, duration: .once)
                        // D01: real original respond resumes before any failed decision save.
                        check(!admitted, "ACP Gate admission: rejected decision reported accepted (\(mode))")
                        // D02
                        check(gate.pending == request && gate.pendingReview == review && gate.queuedCount == 0,
                            "ACP Gate admission: rejected decision discarded its exact card/review (\(mode))")
                        let afterDecision = try sql.load().first { $0.id == submitted.id }
                        let afterJournal = try sql.journal(taskID: submitted.id)
                        // D03
                        check(afterDecision == beforeDecision && afterJournal == beforeJournal
                            && manager.lastPersistenceResult?.primaryCommitted == false,
                            "ACP Gate admission: real decision rejection was not synchronous primary refusal (\(mode))")
                        for _ in 0..<32 { await Task.yield() }
                        // D04
                        check(!FileManager.default.fileExists(atPath: wireFile.path)
                            && ended == 0 && manager.workerBinding(taskID: submitted.id) == binding
                            && PermissionGrantStore.shared.grants == grantsBefore,
                            "ACP Gate admission: rejected decision released wire/worker or changed grants (\(mode))")
                        try sql.withConnection { try TaskStore.exec($0, "DROP TRIGGER reject_gate_decision") }
                        // Both versions follow their REAL branch. Do not invent a second card on original.
                        let retryAccepted: Bool
                        if gate.pending?.id == request.id {
                            retryAccepted = gate.respond(id: request.id, approved: approvalRejected, duration: .once)
                        } else { retryAccepted = false }
                        // D05
                        check(retryAccepted, "ACP Gate admission: exact retained decision failed once after trigger removal (\(mode))")
                    } else if heldMode {
                        let freshSQL = TaskStore(root: directory)
                        holdingSQL = freshSQL
                        let freshStore = AgentTaskStore(fileURL: export, mirror: freshSQL)
                        guard freshStore.allowsHarnessPersistence else { throw FixtureError.unsafeRoot }
                        let fresh = AgentTaskManager(store: freshStore, receiptStore: ActionReceiptStore(fileURL: receiptFile))
                        holdingManager = fresh
                        heldRow = try freshSQL.load().first { $0.id == submitted.id }
                        heldJournal = try freshSQL.journal(taskID: submitted.id)
                        // H01
                        check(fresh.lastPersistenceResult?.primaryCommitted == true && fresh.historyReadFailure == nil
                            && heldRow?.status == .recovering && fresh.backendStartsForTesting == 0
                            && manager.workerBinding(taskID: submitted.id) == binding && captured.isStillValid?() == false,
                            "ACP Gate admission: pending hold was replaced by cancelled task or uncommitted recovery")
                        let admitted = gate.respond(id: request.id, approved: true, duration: .once)
                        // H02
                        check(!admitted && gate.pending == request && gate.pendingReview == review,
                            "ACP Gate admission: stale pending approval discarded card or authorized")
                        for _ in 0..<32 { await Task.yield() }
                        let primaryAfter = try freshSQL.load().first { $0.id == submitted.id }
                        let journalAfter = try freshSQL.journal(taskID: submitted.id)
                        // H03
                        check(primaryAfter == heldRow && journalAfter == heldJournal && ended == 0
                            && !FileManager.default.fileExists(atPath: wireFile.path),
                            "ACP Gate admission: stale pending decision changed held history or released wire")
                        // Explicit durable task stop plus the exact withdrawal seam Session.close uses.
                        // This does not claim an independently invoked Session.cancel callback.
                        fresh.cancel(submitted.id)
                        gate.cancelPending(taskID: submitted.id)
                        expectedApproved = false
                        expectedDecisionKind = .permissionDenied
                        expectedDecisionCount = 0
                        expectedFinalState = .cancelled
                        cancelledForCleanup = true
                        // H04
                        check(fresh.task(id: submitted.id)?.status == .cancelled && gate.pending == nil,
                            "ACP Gate admission: explicit held task stop failed to commit/release false")
                    } else if cancelRejected {
                        gate.cancelPending(taskID: submitted.id)
                        expectedApproved = false
                        expectedDecisionKind = .permissionDenied
                        expectedDecisionCount = 0
                        // X01: denied-journal failure must never strand cancellation's waiter.
                        check(gate.pending == nil && gate.queuedCount == 0
                            && ToolCallReviewStore.shared.review(id: request.id) == nil,
                            "ACP Gate admission: explicit withdrawal stranded card/review after denied-save refusal")
                    } else {
                        // A01
                        check(gate.respond(id: request.id, approved: true, duration: .once),
                            "ACP Gate admission: accepted exact approval was refused")
                    }
                }
                guard await reached({ FileManager.default.fileExists(atPath: wireFile.path) }) else { throw FixtureError.timeout }
                let wire = object(wireFile)
                let result = wire["result"] as? [String: Any] ?? [:]
                let outcome = result["outcome"] as? [String: String]
                // C11
                check(outcome == ["outcome": "selected", "optionId": expectedApproved ? "fixture-allow-once" : "fixture-reject-once"]
                    && result["optionId"] == nil && wire["replyCount"] as? Int == 1,
                    "ACP Gate admission: actual nested wire verdict did not consume admitted decision (\(mode))")
                let peerRequest = object(directory.appendingPathComponent("permission-request.json"))
                let offered = peerRequest["options"] as? [[String: String]] ?? []
                // C21
                check(peerRequest["taskID"] as? String == submitted.id
                    && peerRequest["sessionID"] as? String == sessionID
                    && peerRequest["requestID"] as? Int == 900
                    && offered == [["optionId": "fixture-allow-once", "name": "Allow once", "kind": "allow_once"],
                                   ["optionId": "fixture-reject-once", "name": "Reject", "kind": "reject_once"]],
                    "ACP Gate admission: actual peer lost exact offered options/session/request (\(mode))")
                // C22
                check(object(directory.appendingPathComponent("permission-primary.json"))["state"] as? String == "running"
                    && releaseRow.durability?.attempt == 1 && releaseRow.durability?.receiptIDs == joined,
                    "ACP Gate admission: real nested request bypassed the original running attempt/receipt boundary (\(mode))")
                // C12
                check(wire["requestCount"] as? Int == 1 && wire["promptCount"] as? Int == 1
                    && wire["cancelBeforeVerdict"] as? Bool == false && wire["externalEffectCount"] as? Int == 0,
                    "ACP Gate admission: verdict substituted cancellation/duplicate/effect (\(mode))")
                try JSONSerialization.data(withJSONObject: ["release": true], options: [.sortedKeys])
                    .write(to: directory.appendingPathComponent("release-result.json"), options: .atomic)
                guard await reached({ ended == 1 }) else { throw FixtureError.timeout }
                if cancelRejected {
                    try sql.withConnection { try TaskStore.exec($0, "DROP TRIGGER reject_gate_decision") }
                }
                guard !FileManager.default.fileExists(atPath: directory.appendingPathComponent("peer-error.json").path) else {
                    throw FixtureError.invalidPeer
                }
                let primary = try sql.load().first { $0.id == submitted.id }
                let journal = try sql.journal(taskID: submitted.id)
                let requests = journal.filter { $0.draft.kind == .permissionRequested }
                let decisions = journal.filter { $0.draft.kind == .permissionApproved || $0.draft.kind == .permissionDenied }
                // C13
                check(requests.count == expectedRequestCount && requests.allSatisfy { $0.draft.attempt == 1 }
                    && decisions.count == expectedDecisionCount
                    && decisions.allSatisfy { $0.draft.attempt == 1 && $0.draft.kind == expectedDecisionKind },
                    "ACP Gate admission: actual fresh decision/request journal omitted/duplicated rejected admission (\(mode))")
                // C14
                check(primary?.durability?.attempt == 1 && primary?.durability?.leaseOwner == owner
                    && primary?.durability?.receiptIDs == joined && primary?.status == expectedFinalState,
                    "ACP Gate admission: actual task consumer lost exact attempt/receipt/final primary (\(mode))")
                let persisted = ActionReceiptStore(fileURL: receiptFile)
                guard persisted.receipts.count == 1, let receipt = persisted.receipts.first else { throw FixtureError.missingReceipt }
                // C15
                check(receipt.actionID == fired.actionID && receipt.preparedAction == fired.preparedAction
                    && receipt.taskID == submitted.id && receipt.taskAttempt == 1 && receipt.taskLeaseOwner == owner
                    && receipt.status == .couldNotVerify && receipt.result == marker && receipt.artifacts == [sessionID],
                    "ACP Gate admission: harmless real protocol return lost retained fired/terminal receipt (\(mode))")
                // C16
                check(entered == 1 && ended == 1 && manager.backendStartsForTesting == 1
                    && manager.workerBinding(taskID: submitted.id) == nil && captured.isStillValid?() == false,
                    "ACP Gate admission: actual owned backend failed to join/release exactly once (\(mode))")
                let calls = (try? JSONSerialization.jsonObject(with: Data(contentsOf: callsFile))) as? [String]
                // C17
                check(calls == ["initialize", "session/new", "session/prompt"]
                    && object(directory.appendingPathComponent("launch.json"))["launchCount"] as? Int == 1,
                    "ACP Gate admission: peer protocol relaunched or used cancellation before wire (\(mode))")
                // C18
                check(gate.pending == nil && gate.queuedCount == 0 && PermissionGrantStore.shared.grants == grantsBefore
                    && unexpectedTools == toolsBefore,
                    "ACP Gate admission: actual cleanup/grants/no-backing contract failed (\(mode))")
                if cancelledForCleanup {
                    // H05
                    check(!announcements.contains { $0.contains(marker) }
                        && !journal.contains { $0.draft.kind == .jobCompleted || $0.draft.kind == .permissionApproved },
                        "ACP Gate admission: held cancelled task delivered old success or approval")
                } else {
                    // C19 (nonheld)
                    check(primary?.result == marker + warning && primary?.artifacts == [sessionID]
                        && announcements.contains(marker + warning),
                        "ACP Gate admission: same harmless accepted/rejected protocol reply consumer changed (\(mode))")
                }
                if let requestID { ToolCallReviewStore.shared.remove(id: requestID) }
                let callBytes = try Data(contentsOf: callsFile)
                let freshSQL = TaskStore(root: directory)
                defer { freshSQL.close() }
                let fresh = AgentTaskManager(store: AgentTaskStore(fileURL: export, mirror: freshSQL),
                    receiptStore: ActionReceiptStore(fileURL: receiptFile))
                for _ in 0..<16 { await Task.yield() }
                let rereadPrimary = try freshSQL.load().first { $0.id == submitted.id }
                let rereadJournal = try freshSQL.journal(taskID: submitted.id)
                let callBytesAfter = try Data(contentsOf: callsFile)
                // C20
                check(fresh.task(id: submitted.id) == primary && rereadPrimary == primary && rereadJournal == journal
                    && fresh.backendStartsForTesting == 0 && callBytesAfter == callBytes,
                    "ACP Gate admission: actual fresh reader changed history or replayed peer (\(mode))")
                SelfTest.diagnostic("TASK_ACP_GATE_TRACE: mode=\(mode) wire=\(outcome?["optionId"] ?? "absent") primary=\(primary?.status.rawValue ?? "absent") originalMemory=\(manager.task(id: submitted.id)?.status.rawValue ?? "absent") requests=\(requests.count) decisions=\(decisions.count) starts=\(manager.backendStartsForTesting) ended=\(ended) joined=\(primary?.durability?.receiptIDs.count ?? 0) askDelta=\(gate.askCount - askBefore)")
            } catch {
                try? sql.withConnection { db in
                    try TaskStore.exec(db, "DROP TRIGGER IF EXISTS reject_gate_request")
                    try TaskStore.exec(db, "DROP TRIGGER IF EXISTS reject_gate_decision")
                }
                try? JSONSerialization.data(withJSONObject: ["release": true]).write(to: releaseFile, options: .atomic)
                try? JSONSerialization.data(withJSONObject: ["release": true]).write(
                    to: directory.appendingPathComponent("release-result.json"), options: .atomic)
                PermissionGate.shared.cancelPending(taskID: submitted.id)
                if ended == 0 {
                    manager.cancel(submitted.id)
                    let drained = await reached { ended == 1 }
                    check(drained, "ACP Gate admission: error cleanup failed to drain exact owned worker (\(mode))")
                }
                throw error
            }
        }
        // O01
        check(SelfTestStoreGuard.diff(ownerBefore, SelfTestStoreGuard.take()).isEmpty
            && PermissionGrantStore.shared.grants == grantsBefore,
            "ACP Gate admission: owned fixture changed owner stores/defaults/grants")
    }

    private static let peer = #"""

        root = pathlib.Path(__file__).resolve().parent
        temporary = pathlib.Path(tempfile.gettempdir()).resolve()
        assert root != temporary and temporary in root.parents

        assert pathlib.Path(config['database']).resolve().parent == root
        assert pathlib.Path(config['receipts']).resolve().parent == root
