+                try controller.installNativeTerminationEngineForTesting(empty, stream: nil)
+                try await controller.runNativeTerminationTailForTesting()
+            } catch { invalid = true; safelyQuitInvalid() }
+        }
+    }
+    private func reply(_ reply: NSApplication.TerminateReply) {
+        if firstVerdict { return }
+        check("Q01 actual Cocoa quit replies later", reply == .terminateLater)
+        _ = require("Q02 actual current result returned while retired native pending", currentResultObserved && resultBeforeRetiredEnd)
+    }
+    private func finalReply(_ drained: Bool) -> Bool {
+        guard !firstVerdict else { return drained }
+        firstVerdict = true
+        let at = native.snapshot()
+        check("F01 actual retired native has returned before positive quit", at.active == 0 && at.ends >= 2)
+        check("F02 controller captured same retained native owner", ownerCaptured)
+        check("F03 retained finish bodies have ended before positive quit", ended == assigned)
+        check("C07 actual native success still retained", at.successes >= 1)
+        SelfTest.diagnostic("DIRECT_DICTATION_FIRST_FINAL_REPLY_TRACE mode=\(mode) nativeActive=\(at.active) starts=\(at.starts) ends=\(at.ends) assigned=\(assigned) ended=\(ended) captured=\(ownerCaptured) resultBeforeRetiredEnd=\(resultBeforeRetiredEnd)")
+        // Veto-only safe original cleanup AFTER measured first product verdict;
+        // neither variant can promote false or release entered native early.
+        if at.active != 0 || ended != assigned {
+            vetoes += 1
+            return false
+        }
+        return drained
+    }
+    private func safelyQuitInvalid() {
+        invalid = true
+        // Safety still owns the actual engine on a prerequisite failure; never
+        // accept process exit with an unjoined model API just because INVALID.
+        controller.cancelDictation()
+        SelfTest.requestTermination()
+    }
+    private func safetyDrain() async {
+        // SAME delegate nativeTerminationTask invokes this ONLY after measured
+        // first verdict and actual reply(false). No fixture waiter Task.
+        await retired?.value
+        await current?.value
+        await parent?.value
+        let ended = native.snapshot()
+        if ended.active != 0 || assigned != self.ended {
+            invalid = true
+            SelfTest.diagnostic("DIRECT_DICTATION_TERMINATION_INVALID: unsafe captured cleanup")
+        }
+        SelfTest.diagnostic("DIRECT_DICTATION_CLEANUP_JOIN_TRACE mode=\(mode) nativeActive=\(ended.active) nativeEnds=\(ended.ends) assigned=\(assigned) ended=\(self.ended)")
+    }
+    private func willExit() {
+        let at = native.snapshot()
+        check("J01 actual native API fully returned at real process exit", at.active == 0 && at.starts == at.ends)
+        check("J02 existing finish task bodies fully returned", assigned == ended)
+        check("J03 current result did not wait for retired native", resultBeforeRetiredEnd)
+        check("J04 no owner injection/history calls", output.insertCalls == 0 && output.recordCalls == 0 && output.outcomeCalls == 0)
+        SelfTest.diagnostic("DIRECT_DICTATION_TERMINATION_EXIT_TRACE mode=\(mode) starts=\(at.starts) ends=\(at.ends) successes=\(at.successes) active=\(at.active) assigned=\(assigned) ended=\(ended) vetoes=\(vetoes) checks=\(checks) failures=\(failures) invalid=\(invalid)")
+        SelfTest.diagnostic(invalid ? "DIRECT_DICTATION_TERMINATION_INVALID: prerequisites" : failures == 0 ? "DIRECT_DICTATION_TERMINATION_OK: \(checks) checks" : "DIRECT_DICTATION_TERMINATION_FAILED: \(failures)/\(checks) assertions")
+        try? FileManager.default.removeItem(at: usageRoot)
+    }
+    private func chunk(_ samples: [Float]) throws -> AudioChunk {
+        guard let format = AVAudioFormat(commonFormat: .pcmFormatFloat32, sampleRate: 16_000, channels: 1, interleaved: false),
+              let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: AVAudioFrameCount(samples.count)),
+              let floats = buffer.floatChannelData?[0] else { throw CocoaError(.fileReadCorruptFile) }
+        buffer.frameLength = AVAudioFrameCount(samples.count)
+        samples.withUnsafeBufferPointer { floats.update(from: $0.baseAddress!, count: samples.count) }
+        return AudioChunk(buffer: buffer)
+    }
+}

+/// No retained samples/transcript/model result. Immutable callback plus locked
+/// counters only; entry does not delay or hold the actual vendor operation.
+final class DirectDictationNativeTrace: @unchecked Sendable {
+    struct Snapshot: Sendable { var starts = 0; var ends = 0; var successes = 0; var active = 0 }
+    private let lock = NSLock()
+    private var value = Snapshot()
+    private let onEntry: @Sendable (Int) -> Void
+    init(onEntry: @escaping @Sendable (Int) -> Void) { self.onEntry = onEntry }
+    func entered() {
+        lock.lock(); value.starts += 1; value.active += 1; let ordinal = value.starts; lock.unlock()
+        onEntry(ordinal)
+    }
+    func returned(succeeded: Bool) {
+        lock.lock(); value.ends += 1; value.active -= 1; if succeeded { value.successes += 1 }; lock.unlock()
+    }
+    func snapshot() -> Snapshot { lock.lock(); defer { lock.unlock() }; return value }
+}

+@MainActor
+private final class DirectDictationOutputTrace {
+    var insertCalls = 0
+    var recordCalls = 0
+    var outcomeCalls = 0
+}
