+        check("Q02 actual native work complete before positive quit", observed.active == 0 && observed.starts == observed.ends)

+    }

+    }

+    }

+private final class LateFeedPartials: @unchecked Sendable {
+    private let lock = NSLock()
+    private var createdCount = 0
+    private var late: Task<Void, Never>?
+    private var inputCount = 0
+    private var lastInputNativeActive = -1
+    private var lastInputCancelled = false
+    struct Snapshot: Sendable {
+        let created: Int; let lateTask: Task<Void, Never>?
+        let inputCount: Int; let lastInputNativeActive: Int; let lastInputCancelled: Bool
+    }
+    func inputReceived(activeNative: Int, callerCancelled: Bool) {
+        lock.lock(); defer { lock.unlock() }
+        inputCount += 1; lastInputNativeActive = activeNative; lastInputCancelled = callerCancelled
+    }
+    func created(_ sameTask: Task<Void, Never>) {
+        lock.lock(); defer { lock.unlock() }; createdCount += 1
+        if createdCount == 2 { late = sameTask }
+    }
+    func snapshot() -> Snapshot { lock.lock(); defer { lock.unlock() }; return Snapshot(created: createdCount, lateTask: late, inputCount: inputCount,
+            lastInputNativeActive: lastInputNativeActive, lastInputCancelled: lastInputCancelled) }
+}
