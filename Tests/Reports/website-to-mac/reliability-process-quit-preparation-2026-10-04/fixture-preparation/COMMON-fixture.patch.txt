+    static func isAdmitted(_ value: Admission) -> Bool {
+        guard isRequested, SelfTest.isRunning, admitted == value,
+              value.root.resolvingSymlinksInPath().path == value.root.path,
+              let metadata = try? value.root.resourceValues(forKeys: [.isDirectoryKey, .isSymbolicLinkKey]),
+              metadata.isDirectory == true, metadata.isSymbolicLink != true else { return false }
+        return true
+    }

+    /// Invoked before NSApplication, AppDelegate, private stores or runtime owners.
+    static func validateArgumentsBeforeOwners() throws {
+        guard isRequested, admitted == nil else { throw CocoaError(.fileReadNoPermission) }
+        let args = Array(CommandLine.arguments.dropFirst())
+        let valueFlags: Set<String> = ["--native-background-case", "--native-background-root",
+            SelfTest.outputFlag, SelfTest.timeoutFlag]
+        var values: [String: String] = [:]
+        var seen = Set<String>()
+        var offset = 0
+        while offset < args.count {
+            let flag = args[offset]
+            guard seen.insert(flag).inserted else { throw CocoaError(.fileReadNoPermission) }
+            if flag == "--selftest-native-termination" { offset += 1; continue }
+            guard valueFlags.contains(flag), args.indices.contains(offset + 1),
+                  !args[offset + 1].hasPrefix("-") else { throw CocoaError(.fileReadNoPermission) }
+            values[flag] = args[offset + 1]
+            offset += 2
+        }
+        guard seen.contains("--selftest-native-termination"),
+              values["--native-background-case"] == "continuation-close",
+              let supplied = values["--native-background-root"] else { throw CocoaError(.fileReadNoPermission) }
+        let root = URL(fileURLWithPath: supplied, isDirectory: true).standardizedFileURL
+        let temp = FileManager.default.temporaryDirectory.standardizedFileURL.resolvingSymlinksInPath()
+        let prefix = "NextNotesBackgroundQuit-"
+        guard root.path == supplied, root.deletingLastPathComponent().path == temp.path,
+              root.lastPathComponent.hasPrefix(prefix),
+              UUID(uuidString: String(root.lastPathComponent.dropFirst(prefix.count))) != nil,
+              root.resolvingSymlinksInPath().path == root.path,
+              !FileManager.default.fileExists(atPath: root.path),
+              (try? FileManager.default.attributesOfItem(atPath: root.path)) == nil else {
+            throw CocoaError(.fileReadNoPermission)
+        }
+        if let output = values[SelfTest.outputFlag] {
+            guard output == root.appendingPathComponent("out.log").path else { throw CocoaError(.fileReadNoPermission) }
+        }
+        if let timeout = values[SelfTest.timeoutFlag] {
+            guard let seconds = Double(timeout), seconds.isFinite, seconds > 0, seconds <= 300 else {
+                throw CocoaError(.fileReadNoPermission)
+            }
+        }
+        try FileManager.default.createDirectory(at: root, withIntermediateDirectories: false)
+        let value = Admission(root: root, nonce: UUID())
+        admitted = value
+        guard isAdmitted(value) else { admitted = nil; throw CocoaError(.fileReadNoPermission) }
+        // Any seeding/strict-reader failure is rejected before NSApplication exists.
+        preparedFixture = try Fixture(admission: value)
+    }
+    private static func refuseBootstrap() -> Never {
+        SelfTest.failed = true
+        SelfTest.diagnostic("BACKGROUND_QUIT_INVALID: private bootstrap unavailable before native quit")
+        exit(1) // This model-free mode has entered no worker/model/native operation.
+    }

+    static func armIfRequested() async throws -> Bool {
+        guard isRequested else { return false }
+        guard let admission = admitted, isAdmitted(admission), let fixture = preparedFixture,
+              let delegate = AppDelegate.current else { refuseBootstrap() }
+        let empty = EmbeddingRuntime(modelURL: nil, displayName: "Empty diagnostic quit owner",
+            tag: "empty-background-quit", notesModelBusy: { false }, mayLoad: { _ in false })
+        let notes = NotesModelRuntime(spec: ModelSpec(displayName: "Empty diagnostic quit owner",
+            fileName: "never-open.gguf", url: admission.root.appendingPathComponent("never-open.gguf"),
+            expectedBytes: 0, expectedSHA256: nil), gpuLayers: 0, openFailureSink: nil,
+            pressureSnapshot: { .init(level: .normal, generation: 0, changedAt: Date(timeIntervalSince1970: 0)) })
+        do {
+            try delegate.configureBackgroundTerminationFixture(
+                manager: fixture.manager, runtime: notes, embedding: empty, admission: admission,
+                reply: { fixture.observe($0) },
+                finalReply: { fixture.finalReply($0) },
+                safetyDrain: { fixture.afterFirstVeto() },
+                willExit: { fixture.willExit() }
+            )
+        } catch { refuseBootstrap() }
+        // Existing run-loop quit admission; no fixed delay or MainActor blocking.
+        SelfTest.diagnostic("BACKGROUND_QUIT_INPUT_OK: retained private pending rows; no live worker or native backing")
+        SelfTest.requestTermination()
+        return true

+            let fixtureRows = [

+                    status: .waitingForInput, tool: "filesystem.read", pendingInteraction: .input(fixtureInput)),

+                    status: .waitingForPermission, tool: "filesystem.read", arguments: fixtureApproval.arguments,

+        func finalReply(_ drained: Bool) -> Bool {

+            check(replies >= 1, "actual later reply preceded final proposal")
+            if finals == 1 { return false } // Veto-only; same App task owns safetyDrain.
+            check(cleanedVeto && finals == 2, "one-shot same-task safety drain precedes second proposal")
+            return drained
+        }
+        func afterFirstVeto() {
+            check(finals == 1 && !cleanedVeto, "safety continuation consumed exactly once")
+            cleanedVeto = true
+        }
+        func willExit() {
+            check(finals == 2 && replies == 2 && cleanedVeto, "actual Cocoa retry and willTerminate reached once")

+                check(snapshot.closed && snapshot.runningIDs.isEmpty && snapshot.backendStarts == 0,
+                    "empty registered Manager remains closed at physical process exit callback")
+            } else { invalid = true; SelfTest.failed = true }
+            sql.close()
+            // Retain ONLY private fixture files for the serial owner's report; no owner data.
+            if invalid { SelfTest.diagnostic("BACKGROUND_QUIT_INVALID: " + String(checks) + " checks; contract input unavailable") }
+            else if failures == 0 { SelfTest.diagnostic("BACKGROUND_QUIT_OK: " + String(checks) + " checks; continuation-close only") }
+            else { SelfTest.diagnostic("BACKGROUND_QUIT_FAILED: " + String(failures) + "/" + String(checks)) }
+        }
